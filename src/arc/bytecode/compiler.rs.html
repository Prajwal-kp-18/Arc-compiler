1//! # Bytecode Compiler
2//!
3//! Lowers a Resolver-annotated AST into a [`CompiledProgram`] of [`Chunk`]s.
4//! No execution happens here — see the Phase 2 design doc
5//! (`local-notes/phase2-bytecode-design.md`) for the full rationale.
6//!
7//! Like the evaluator, this is infallible: it trusts the Resolver's
8//! guarantees completely and reads `SlotIndex`/`FunctionId`/`ResolvedBinding`/
9//! `ResolvedCall` straight off the AST's `Cell` fields instead of
10//! re-deriving scope.
11
12use crate::ast::resolver::{BuiltinFn, ResolvedBinding, ResolvedCall};
13use crate::ast::types::Value;
14use crate::ast::{
15    Ast, ASTBinaryExpression, ASTBinaryOperatorKind, ASTExpression, ASTExpressionKind,
16    ASTFunctionCallExpression, ASTFunctionDeclaration, ASTIfStatement, ASTPostfixUnaryExpression,
17    ASTStatement, ASTStatementKind, ASTUnaryExpression, ASTUnaryOperatorKind,
18};
19use crate::bytecode::chunk::Chunk;
20use crate::bytecode::opcode::OpCode;
21
22/// The output of a compile pass: the top-level "script" chunk, plus one
23/// chunk per function, indexed by `FunctionId.0`.
24pub struct CompiledProgram {
25    pub script: Chunk,
26    pub functions: Vec<Chunk>,
27}
28
29pub struct BytecodeCompiler {
30    functions: Vec<Chunk>,
31    /// In-progress chunks; the top is the one currently being written to.
32    /// Depth > 1 only while compiling a function nested inside another.
33    chunk_stack: Vec<Chunk>,
34    /// Source byte-offset of the most recently seen token, reused for AST
35    /// nodes that don't carry their own (e.g. number literals) so
36    /// disassembly still points somewhere close to the right place.
37    current_offset: u32,
38}
39
40impl BytecodeCompiler {
41    pub fn new(function_count: u32) -> Self {
42        Self {
43            functions: (0..function_count).map(|_| Chunk::new()).collect(),
44            chunk_stack: Vec::new(),
45            current_offset: 0,
46        }
47    }
48
49    pub fn compile(mut self, ast: &Ast) -> CompiledProgram {
50        self.chunk_stack.push(Chunk::new());
51        self.compile_body(&ast.statements);
52        let script = self.chunk_stack.pop().expect("script chunk was just pushed");
53        CompiledProgram { script, functions: self.functions }
54    }
55
56    fn current(&mut self) -> &mut Chunk {
57        self.chunk_stack.last_mut().expect("always compiling inside some chunk")
58    }
59
60    fn emit(&mut self, op: OpCode, offset: u32) {
61        self.current().write_op(op, offset);
62    }
63
64    fn emit_constant(&mut self, value: Value, offset: u32) {
65        let idx = self.current().add_constant(value);
66        self.current().write_op(OpCode::Constant, offset);
67        self.current().write_u16(idx, offset);
68    }
69
70    fn slot_u16(index: u32) -> u16 {
71        u16::try_from(index).expect("slot index exceeded u16::MAX")
72    }
73
74    fn emit_get(&mut self, binding: ResolvedBinding, offset: u32) {
75        let (op, slot) = match binding {
76            ResolvedBinding::Local(s) => (OpCode::GetLocal, s.0),
77            ResolvedBinding::Global(s) => (OpCode::GetGlobal, s.0),
78        };
79        self.current().write_op(op, offset);
80        self.current().write_u16(Self::slot_u16(slot), offset);
81    }
82
83    fn emit_set(&mut self, binding: ResolvedBinding, offset: u32) {
84        let (op, slot) = match binding {
85            ResolvedBinding::Local(s) => (OpCode::SetLocal, s.0),
86            ResolvedBinding::Global(s) => (OpCode::SetGlobal, s.0),
87        };
88        self.current().write_op(op, offset);
89        self.current().write_u16(Self::slot_u16(slot), offset);
90    }
91
92    /// Placeholder pushed where there's genuinely no value to give (an
93    /// implicit return with no trailing expression statement, or `print`'s
94    /// result) — see design doc §4/§5. Reuses the evaluator's inert
95    /// `Integer(0)` filler convention for consistency.
96    fn emit_placeholder(&mut self, offset: u32) {
97        self.emit_constant(Value::Integer(0), offset);
98    }
99
100    // -- statements -----------------------------------------------------------
101
102    /// Compiles a function/script body: every statement is stack-neutral
103    /// except (optionally) the last. If the last statement is an expression
104    /// statement, its value becomes the implicit return (no trailing `POP`);
105    /// otherwise a placeholder is pushed. Either way, always ends in
106    /// `OP_RETURN` — every chunk needs one, explicit `return` or not (see
107    /// design doc §4).
108    fn compile_body(&mut self, statements: &[ASTStatement]) {
109        let Some((last, rest)) = statements.split_last() else {
110            let offset = self.current_offset;
111            self.emit_placeholder(offset);
112            self.emit(OpCode::Return, offset);
113            return;
114        };
115
116        for stmt in rest {
117            self.compile_statement(stmt);
118        }
119
120        if let ASTStatementKind::Expression(expr) = &last.kind {
121            self.compile_expression(expr);
122        } else {
123            self.compile_statement(last);
124            self.emit_placeholder(self.current_offset);
125        }
126        self.emit(OpCode::Return, self.current_offset);
127    }
128
129    fn compile_statements(&mut self, statements: &[ASTStatement]) {
130        for stmt in statements {
131            self.compile_statement(stmt);
132        }
133    }
134
135    fn compile_statement(&mut self, stmt: &ASTStatement) {
136        match &stmt.kind {
137            ASTStatementKind::Expression(expr) => {
138                self.compile_expression(expr);
139                self.emit(OpCode::Pop, self.current_offset);
140            }
141            ASTStatementKind::VariableDeclaration(decl) => {
142                self.compile_expression(&decl.initializer);
143                if let Some(token) = &decl.name_span {
144                    self.current_offset = token.span.start as u32;
145                }
146                let binding = decl.binding.get().expect("Resolver guarantees every declaration is resolved");
147                self.emit_set(binding, self.current_offset);
148                self.emit(OpCode::Pop, self.current_offset);
149            }
150            ASTStatementKind::Assignment(assign) => {
151                self.compile_expression(&assign.value);
152                if let Some(token) = &assign.name_span {
153                    self.current_offset = token.span.start as u32;
154                }
155                if assign.needs_float_widen.get() {
156                    self.emit(OpCode::ToFloat, self.current_offset);
157                }
158                let binding = assign.binding.get().expect("Resolver guarantees every assignment target is resolved");
159                self.emit_set(binding, self.current_offset);
160                self.emit(OpCode::Pop, self.current_offset);
161            }
162            ASTStatementKind::Block(statements) => self.compile_statements(statements),
163            ASTStatementKind::IfStatement(if_stmt) => self.compile_if(if_stmt),
164            ASTStatementKind::FunctionDeclaration(func) => self.compile_function(func),
165            ASTStatementKind::ReturnStatement(ret) => {
166                self.compile_expression(&ret.value);
167                self.emit(OpCode::Return, self.current_offset);
168            }
169        }
170    }
171
172    fn compile_if(&mut self, if_stmt: &ASTIfStatement) {
173        self.compile_expression(&if_stmt.condition);
174        let offset = self.current_offset;
175
176        let then_jump = self.current().emit_jump(OpCode::JumpIfFalse, offset);
177        self.emit(OpCode::Pop, offset);
178        self.compile_statements(&if_stmt.then_branch);
179        let else_jump = self.current().emit_jump(OpCode::Jump, offset);
180
181        self.current().patch_jump(then_jump);
182        self.emit(OpCode::Pop, offset);
183        if let Some(else_branch) = &if_stmt.else_branch {
184            self.compile_statements(else_branch);
185        }
186
187        self.current().patch_jump(else_jump);
188    }
189
190    fn compile_function(&mut self, func: &ASTFunctionDeclaration) {
191        let id = func.function_id.get().expect("Resolver assigns every function an id");
192        self.chunk_stack.push(Chunk::new());
193        self.compile_body(&func.body);
194        let chunk = self.chunk_stack.pop().expect("function chunk was just pushed");
195        self.functions[id.0 as usize] = chunk;
196    }
197
198    // -- expressions ------------------------------------------------------------
199
200    fn compile_expression(&mut self, expr: &ASTExpression) {
201        match &expr.kind {
202            ASTExpressionKind::Number(n) => {
203                let offset = self.current_offset;
204                match n.value {
205                    Value::Boolean(true) => self.emit(OpCode::True, offset),
206                    Value::Boolean(false) => self.emit(OpCode::False, offset),
207                    ref v => self.emit_constant(v.clone(), offset),
208                }
209            }
210            ASTExpressionKind::Binary(bin) => self.compile_binary(bin),
211            ASTExpressionKind::Paranthesized(p) => self.compile_expression(&p.expression),
212            ASTExpressionKind::Unary(u) => self.compile_unary(u),
213            ASTExpressionKind::PostfixUnary(p) => self.compile_postfix(p),
214            ASTExpressionKind::Identifier(ident) => {
215                if let Some(token) = &ident.token {
216                    self.current_offset = token.span.start as u32;
217                }
218                let binding = ident.binding.get().expect("Resolver guarantees every identifier is resolved");
219                self.emit_get(binding, self.current_offset);
220            }
221            ASTExpressionKind::FunctionCall(call) => self.compile_call(call),
222        }
223    }
224
225    fn binary_opcode(kind: &ASTBinaryOperatorKind) -> OpCode {
226        use ASTBinaryOperatorKind::*;
227        match kind {
228            Plus => OpCode::Add,
229            Minus => OpCode::Subtract,
230            Multiply => OpCode::Multiply,
231            Divide => OpCode::Divide,
232            Modulo => OpCode::Modulo,
233            Exponentiation => OpCode::Power,
234            BitwiseAnd => OpCode::BitAnd,
235            BitwiseOr => OpCode::BitOr,
236            BitwiseXor => OpCode::BitXor,
237            LeftShift => OpCode::ShiftLeft,
238            RightShift => OpCode::ShiftRight,
239            Equal => OpCode::Equal,
240            NotEqual => OpCode::NotEqual,
241            Less => OpCode::Less,
242            Greater => OpCode::Greater,
243            LessEqual => OpCode::LessEqual,
244            GreaterEqual => OpCode::GreaterEqual,
245            LogicalAnd | LogicalOr => unreachable!("handled directly in compile_binary"),
246        }
247    }
248
249    /// `&&`/`||` compile to jumps + double-`NOT` rather than dedicated
250    /// opcodes: Arc's logical operators always coerce their result to a real
251    /// `Boolean` (unlike Lox-style "return whichever operand"), and `NOT NOT`
252    /// performs exactly that coercion on whichever value survives on the
253    /// stack. See design doc §3.
254    fn compile_binary(&mut self, bin: &ASTBinaryExpression) {
255        self.current_offset = bin.operator.token.span.start as u32;
256        let offset = self.current_offset;
257
258        match bin.operator.kind {
259            ASTBinaryOperatorKind::LogicalAnd => {
260                self.compile_expression(&bin.left);
261                let end_jump = self.current().emit_jump(OpCode::JumpIfFalse, offset);
262                self.emit(OpCode::Pop, offset);
263                self.compile_expression(&bin.right);
264                self.current().patch_jump(end_jump);
265                self.emit(OpCode::Not, offset);
266                self.emit(OpCode::Not, offset);
267            }
268            ASTBinaryOperatorKind::LogicalOr => {
269                self.compile_expression(&bin.left);
270                let else_jump = self.current().emit_jump(OpCode::JumpIfFalse, offset);
271                let end_jump = self.current().emit_jump(OpCode::Jump, offset);
272                self.current().patch_jump(else_jump);
273                self.emit(OpCode::Pop, offset);
274                self.compile_expression(&bin.right);
275                self.current().patch_jump(end_jump);
276                self.emit(OpCode::Not, offset);
277                self.emit(OpCode::Not, offset);
278            }
279            ref kind => {
280                self.compile_expression(&bin.left);
281                self.compile_expression(&bin.right);
282                self.emit(Self::binary_opcode(kind), offset);
283            }
284        }
285    }
286
287    fn compile_unary(&mut self, u: &ASTUnaryExpression) {
288        self.current_offset = u.operator.token.span.start as u32;
289        let offset = self.current_offset;
290        match u.operator.kind {
291            ASTUnaryOperatorKind::Increment | ASTUnaryOperatorKind::Decrement => {
292                self.compile_inc_dec(&u.operand, matches!(u.operator.kind, ASTUnaryOperatorKind::Increment), false, offset);
293            }
294            // Runtime passthrough: valid for every type, never errors, so
295            // there's nothing to emit at all.
296            ASTUnaryOperatorKind::Plus => self.compile_expression(&u.operand),
297            ASTUnaryOperatorKind::Minus => {
298                self.compile_expression(&u.operand);
299                self.emit(OpCode::Negate, offset);
300            }
301            ASTUnaryOperatorKind::LogicalNot => {
302                self.compile_expression(&u.operand);
303                self.emit(OpCode::Not, offset);
304            }
305        }
306    }
307
308    fn compile_postfix(&mut self, p: &ASTPostfixUnaryExpression) {
309        self.current_offset = p.operator.token.span.start as u32;
310        let offset = self.current_offset;
311        let is_increment = matches!(p.operator.kind, ASTUnaryOperatorKind::Increment);
312        self.compile_inc_dec(&p.operand, is_increment, true, offset);
313    }
314
315    /// Desugars `++x`/`--x`/`x++`/`x--` into `GET`/`CONSTANT 1`/`ADD`-or-
316    /// `SUBTRACT`/`SET` (plus a `POP` for postfix, to discard the new value
317    /// and keep the old one), reusing the generic runtime arithmetic instead
318    /// of dedicated opcodes.
319    ///
320    /// ponytail: this means a String/Boolean value flowing through an
321    /// untyped (`Any`) variable that gets incremented will error as "Cannot
322    /// add" rather than "Cannot increment" once a VM exists. Every *typed*
323    /// case is already rejected at resolve time, so only that narrow
324    /// dynamic-fallback sliver is affected — an accepted trade-off, not an
325    /// oversight (design doc §3).
326    fn compile_inc_dec(&mut self, operand: &ASTExpression, is_increment: bool, is_postfix: bool, offset: u32) {
327        let ASTExpressionKind::Identifier(ident) = &operand.kind else {
328            unreachable!("Resolver guarantees increment/decrement operand is an identifier");
329        };
330        let binding = ident.binding.get().expect("Resolver guarantees this identifier is resolved");
331
332        if is_postfix {
333            self.emit_get(binding, offset); // old value: kept as the final result
334        }
335        self.emit_get(binding, offset);
336        self.emit_constant(Value::Integer(1), offset);
337        self.emit(if is_increment { OpCode::Add } else { OpCode::Subtract }, offset);
338        self.emit_set(binding, offset); // leaves the new value on the stack
339        if is_postfix {
340            self.emit(OpCode::Pop, offset); // discard new value; old value remains
341        }
342    }
343
344    fn compile_call(&mut self, call: &ASTFunctionCallExpression) {
345        if let Some(token) = &call.token {
346            self.current_offset = token.span.start as u32;
347        }
348        let offset = self.current_offset;
349
350        for arg in &call.arguments {
351            self.compile_expression(arg);
352        }
353        let argc = u8::try_from(call.arguments.len()).expect("more than 255 arguments in a single call");
354
355        match call.resolved_call.get().expect("Resolver guarantees every call is resolved") {
356            ResolvedCall::User(id) => {
357                self.emit(OpCode::Call, offset);
358                self.current().write_u16(Self::slot_u16(id.0), offset);
359                self.current().write_u8(argc, offset);
360            }
361            ResolvedCall::Builtin(builtin) => {
362                let op = match builtin {
363                    BuiltinFn::Print => OpCode::Print,
364                    BuiltinFn::Max => OpCode::Max,
365                    BuiltinFn::Min => OpCode::Min,
366                };
367                self.emit(op, offset);
368                self.current().write_u8(argc, offset);
369            }
370        }
371    }

382    /// Lexes, parses, resolves, and compiles a program, panicking on any
383    /// parse/resolve error (these tests are about the bytecode compiler,
384    /// not upstream diagnostics).
385    fn compile(source: &str) -> CompiledProgram {
386        let mut lexer = Lexer::new(source);
387        let mut tokens = Vec::new();
388        while let Some(token) = lexer.next_token() {
389            tokens.push(token);
390        }
391        let mut parser = Parser::new(tokens);
392        let mut ast = Ast::new();
393        while let Some(stmt) = parser.next_statement() {
394            ast.add_statement(stmt);
395        }
396        assert!(
397            parser.diagnostics.is_empty(),
398            "parse errors: {:?}",
399            parser.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>()
400        );
401
402        let mut resolver = Resolver::new();
403        resolver.resolve(&ast);
404        assert!(
405            !resolver.has_errors(),
406            "resolve errors: {:?}",
407            resolver.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>()
408        );
409
410        BytecodeCompiler::new(resolver.function_count()).compile(&ast)
411    }

413    fn op_lines(text: &str) -> Vec<&str> {
414        text.lines().filter(|l| l.contains("OP_")).collect()
415    }

417    #[test]
418    fn test_arithmetic_compiles_and_disassembles() {
419        let program = compile("1 + 2 * 3;");
420        let text = disassemble_chunk(&program.script, "script");
421        assert!(text.contains("OP_CONSTANT"));
422        assert!(text.contains("OP_MULTIPLY"));
423        assert!(text.contains("OP_ADD"));
424        assert!(text.contains("OP_RETURN"));
425        // Last statement is an expression statement -> falls through to
426        // RETURN with no trailing POP (design doc §4).
427        assert!(!text.contains("OP_POP"));
428    }

493    #[test]
494    fn test_print_of_print_stack_effect() {
495        // The narrowest case Arc's grammar allows for a call's result being
496        // consumed by another call: print(print("x")). The outer PRINT must
497        // consume exactly the inner call's placeholder result — no extra
498        // push should appear between the two PRINT instructions (design doc §5).

503        assert!(lines[0].contains("OP_CONSTANT"));
504        assert!(lines[1].contains("OP_PRINT"));
505        assert!(lines[2].contains("OP_PRINT"));
506        assert!(lines[3].contains("OP_RETURN"));
