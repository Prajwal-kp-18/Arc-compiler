1//! # OpCode
2//!
3//! The bytecode instruction set. Deliberately small and generic — every
4//! arithmetic/comparison op dispatches dynamically over the same tagged
5//! [`Value`](crate::ast::types::Value) enum the tree-walking evaluator uses.
6//! Type-specialized instructions are a Phase 5/LLVM concern, not this one.
7//!
8//! Operand widths (bytes immediately following the opcode byte in a
9//! [`Chunk`](crate::bytecode::chunk::Chunk)'s code):
10//!
11//! | OpCode | Operands |
12//! |---|---|
13//! | `Constant` | `u16` constant pool index |
14//! | `GetLocal` / `SetLocal` / `GetGlobal` / `SetGlobal` | `u16` slot index |
15//! | `Jump` / `JumpIfFalse` | `u16` forward offset (added to ip after the operand) |
16//! | `Call` | `u16` function id, `u8` argument count |
17//! | `Print` / `Max` / `Min` | `u8` argument count |
18//! | everything else | none |
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20#[repr(u8)]
21pub enum OpCode {
22    /// Push `constants[operand]`.
23    Constant = 0,
24    /// Push `Value::Boolean(true)` / `Value::Boolean(false)` directly, no constant pool entry.
25    True = 1,
26    False = 2,
27    /// Discard the top of the stack (a statement's leftover expression value).
28    Pop = 3,
29    GetLocal = 4,
30    /// Store into a local slot; leaves the stored value on the stack
31    /// (assignment-as-expression convention, used by inc/dec desugaring).
32    SetLocal = 5,
33    GetGlobal = 6,
34    SetGlobal = 7,
35    /// Convert the top of the stack from `Integer` to `Float` (a no-op if
36    /// already `Float`). Emitted only where the Resolver statically decided
37    /// an assignment needs int->float widening — the *decision* is made at
38    /// compile time; this opcode just performs the runtime conversion.
39    ToFloat = 8,
40    Add = 9,
41    Subtract = 10,
42    Multiply = 11,
43    Divide = 12,
44    Modulo = 13,
45    Power = 14,
46    BitAnd = 15,
47    BitOr = 16,
48    BitXor = 17,
49    ShiftLeft = 18,
50    ShiftRight = 19,
51    Equal = 20,
52    NotEqual = 21,
53    Less = 22,
54    Greater = 23,
55    LessEqual = 24,
56    GreaterEqual = 25,
57    /// Arithmetic negation (`-x`). Unary `+` is a runtime no-op and compiles
58    /// to nothing at all, so it has no opcode.
59    Negate = 26,
60    /// Logical NOT, coercing the operand to boolean first (`!x`).
61    Not = 27,
62    /// Unconditional relative forward jump.
63    Jump = 28,
64    /// Peeks (does not pop) the top of the stack; jumps forward if falsy.
65    JumpIfFalse = 29,
66    /// Calls a user-defined function: `u16` FunctionId, `u8` argument count.
67    /// Arguments are already on the stack, pushed left-to-right.
68    Call = 30,
69    /// Ends the current function/script chunk; the return value is
70    /// whatever's on top of the stack.
71    Return = 31,
72    /// Built-in `print(...)`: `u8` argument count. Pops the arguments and
73    /// pushes an inert placeholder (see design doc §5) since print has no
74    /// real return value.
75    Print = 32,
76    Max = 33,
77    Min = 34,
78}
79
80impl OpCode {
81    pub fn from_byte(byte: u8) -> Option<OpCode> {
82        use OpCode::*;
83        Some(match byte {
84            0 => Constant,
85            1 => True,
86            2 => False,
87            3 => Pop,
88            4 => GetLocal,
89            5 => SetLocal,
90            6 => GetGlobal,
91            7 => SetGlobal,
92            8 => ToFloat,
93            9 => Add,
94            10 => Subtract,
95            11 => Multiply,
96            12 => Divide,
97            13 => Modulo,
98            14 => Power,
99            15 => BitAnd,
100            16 => BitOr,
101            17 => BitXor,
102            18 => ShiftLeft,
103            19 => ShiftRight,
104            20 => Equal,
105            21 => NotEqual,
106            22 => Less,
107            23 => Greater,
108            24 => LessEqual,
109            25 => GreaterEqual,
110            26 => Negate,
111            27 => Not,
112            28 => Jump,
113            29 => JumpIfFalse,
114            30 => Call,
115            31 => Return,
116            32 => Print,
117            33 => Max,
118            34 => Min,
119            _ => return None,
120        })
121    }

123    /// Human-readable mnemonic, used by the disassembler.

125        use OpCode::*;

127            Constant => "OP_CONSTANT",
128            True => "OP_TRUE",
129            False => "OP_FALSE",
130            Pop => "OP_POP",
131            GetLocal => "OP_GET_LOCAL",
132            SetLocal => "OP_SET_LOCAL",
133            GetGlobal => "OP_GET_GLOBAL",
134            SetGlobal => "OP_SET_GLOBAL",
135            ToFloat => "OP_TO_FLOAT",
136            Add => "OP_ADD",
137            Subtract => "OP_SUBTRACT",
138            Multiply => "OP_MULTIPLY",
139            Divide => "OP_DIVIDE",
