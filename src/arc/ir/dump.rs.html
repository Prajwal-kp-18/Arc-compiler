1//! # IR Dump
2//!
3//! Text pretty-printer for [`IrProgram`] — the disassembler's sibling,
4//! wired up behind `--dump-ir` (post-lowering) and `--dump-ir=opt`
5//! (post-passes). The before/after diff of these two is the Phase 4 exit
6//! criterion's demo artifact. [`dump_liveness`] is the same idea for Phase
7//! 4.6's slot liveness analysis, behind `--dump-liveness`.
8

17pub fn dump_program(program: &IrProgram) -> String {
18    let mut out = dump_function(&program.script, "script");
19    for (id, function) in program.functions.iter().enumerate() {
20        out.push('\n');
21        out.push_str(&dump_function(function, &format!("fn#{} {}", id, function.name)));
22    }
23    out
24}
25
26pub fn dump_function(f: &IrFunction, name: &str) -> String {
27    let mut out = format!("== {} (frame_size={}, {} instrs) ==\n", name, f.frame_size, f.instr_count());
28    for (id, block) in f.blocks.iter().enumerate() {
29        if block.dead {
30            continue;
31        }
32        out.push_str(&format!("b{}:\n", id));
33        for instr in &block.instrs {
34            out.push_str(&format!("    {}\n", render(&instr.kind)));
35        }
36        let term = match &block.terminator {
37            Terminator::Jump(b) => format!("jump b{}", b.0),
38            Terminator::Branch { cond, then_block, else_block } => {
39                format!("branch r{} ? b{} : b{}", cond.0, then_block.0, else_block.0)
40            }
41            Terminator::Return(r) => format!("return r{}", r.0),
42        };
43        out.push_str(&format!("    {}\n", term));
44    }
45    out
46}

48/// Prints each function's CFG annotated with slot liveness: every block's
49/// live-in/live-out sets and every instruction's live-after set. `--dump-ir`
50/// output is untouched by this — a separate printer behind `--dump-liveness`.
51pub fn dump_liveness(program: &IrProgram) -> String {
52    let mut out = dump_function_liveness(&program.script, "script");
53    for (id, function) in program.functions.iter().enumerate() {
54        out.push('\n');
55        out.push_str(&dump_function_liveness(function, &format!("fn#{} {}", id, function.name)));
56    }
57    out
58}

60fn dump_function_liveness(f: &IrFunction, name: &str) -> String {
61    let live = liveness::analyze(f);
62    let mut out = format!("== {} (frame_size={}, {} instrs) ==\n", name, f.frame_size, f.instr_count());
63    if f.frame_size == 0 {
64        out.push_str("-- no locals in this frame; liveness only tracks LoadLocal/StoreLocal, not globals --\n");
65    }
66    for (id, block) in f.blocks.iter().enumerate() {
67        if block.dead {
68            continue;
69        }
70        out.push_str(&format!("b{}: live-in={}\n", id, fmt_slots(&live.live_in[id])));
71        let live_after = liveness::live_after_each(block, &live.live_out[id]);
72        for (instr, after) in block.instrs.iter().zip(live_after.iter()) {
73            out.push_str(&format!("    {}    ; live-after={}\n", render(&instr.kind), fmt_slots(after)));
74        }
75        let term = match &block.terminator {
76            Terminator::Jump(b) => format!("jump b{}", b.0),
77            Terminator::Branch { cond, then_block, else_block } => {
78                format!("branch r{} ? b{} : b{}", cond.0, then_block.0, else_block.0)
79            }
80            Terminator::Return(r) => format!("return r{}", r.0),
81        };
82        out.push_str(&format!("    {}\n", term));
83        out.push_str(&format!("b{}: live-out={}\n", id, fmt_slots(&live.live_out[id])));
84    }
85    out
86}

88fn fmt_slots(slots: &HashSet<u32>) -> String {
89    let mut sorted: Vec<u32> = slots.iter().copied().collect();
90    sorted.sort_unstable();
91    format!("{{{}}}", sorted.iter().map(|s| s.to_string()).collect::<Vec<_>>().join(","))
92}

94fn render(kind: &InstrKind) -> String {
95    use InstrKind::*;
96    match kind {
97        Const { dst, value } => format!("r{} = const {:?}", dst.0, value),
98        Binary { dst, op, lhs, rhs, can_error } => format!(
99            "r{} = {} r{}, r{}{}",
100            dst.0,
101            mnemonic(op),
102            lhs.0,
103            rhs.0,
104            if *can_error { " !" } else { "" }
105        ),
106        Negate { dst, src, can_error } => {
107            format!("r{} = neg r{}{}", dst.0, src.0, if *can_error { " !" } else { "" })
108        }
109        Not { dst, src } => format!("r{} = not r{}", dst.0, src.0),
110        ToBool { dst, src } => format!("r{} = to_bool r{}", dst.0, src.0),
111        ToFloat { dst, src } => format!("r{} = to_float r{}", dst.0, src.0),
112        LoadLocal { dst, slot } => format!("r{} = load_local {}", dst.0, slot.0),
113        StoreLocal { slot, src } => format!("store_local {} = r{}", slot.0, src.0),
114        LoadGlobal { dst, slot } => format!("r{} = load_global {}", dst.0, slot.0),
115        StoreGlobal { slot, src } => format!("store_global {} = r{}", slot.0, src.0),
116        Call { dst, function, args } => format!("r{} = call fn#{}({})", dst.0, function, regs(args)),
117        CallBuiltin { dst, builtin, args } => {
118            let name = match builtin {
119                BuiltinFn::Print => "print",
120                BuiltinFn::Max => "max",
121                BuiltinFn::Min => "min",
122            };
123            format!("r{} = call {}({})", dst.0, name, regs(args))
124        }
125    }
126}
127
128fn regs(args: &[super::instr::Reg]) -> String {
129    args.iter().map(|r| format!("r{}", r.0)).collect::<Vec<_>>().join(", ")
130}

132fn mnemonic(op: &ASTBinaryOperatorKind) -> &'static str {
133    use ASTBinaryOperatorKind::*;
134    match op {
135        Plus => "add",
136        Minus => "sub",
137        Multiply => "mul",
138        Divide => "div",
139        Modulo => "mod",
140        Exponentiation => "pow",
141        BitwiseAnd => "band",
142        BitwiseOr => "bor",
143        BitwiseXor => "bxor",
144        LeftShift => "shl",
145        RightShift => "shr",
146        Equal => "eq",
147        NotEqual => "ne",
148        Less => "lt",
149        Greater => "gt",
150        LessEqual => "le",
151        GreaterEqual => "ge",
152        LogicalAnd | LogicalOr => unreachable!("logical operators lower to control flow"),
153    }
154}
