1//! # IR → Bytecode Lowering
2//!
3//! Makes the IR *runnable*: lowers an (optimized) [`IrProgram`] to the same
4//! `Chunk`s the Phase 2 compiler emits, so the existing VM executes it and
5//! the golden suite gates every optimization pass. Wired up behind `--opt`.
6//!
7//! Register scheme: spill-everything. Each virtual register gets a frame
8//! slot above the function's real locals (`slot = frame_size + reg`), every
9//! instruction loads its operands from slots and stores its result back, and
10//! the operand stack is empty at every instruction boundary. Deliberately
11//! naive — correctness is the requirement; the VM doesn't care, and Phase 5
12//! consumes the IR directly, not this bytecode.
13//!
14//! Jumps: block creation order in `lower.rs` guarantees every CFG edge
15//! targets a higher block id, and blocks are emitted in id order, so the
16//! forward-only `Jump`/`JumpIfFalse` opcodes suffice. // CFG is a DAG today
17
18use crate::ast::resolver::{BuiltinFn, SlotIndex};
19use crate::bytecode::chunk::Chunk;
20use crate::bytecode::compiler::{CompiledFunction, CompiledProgram};
21use crate::bytecode::opcode::OpCode;
22
23use super::instr::{Block, BlockId, InstrKind, IrFunction, IrProgram, Reg, Terminator};

25pub fn program_to_bytecode(program: &IrProgram) -> CompiledProgram {
26    let script = function_to_chunk(&program.script);
27    let functions = program
28        .functions
29        .iter()
30        .map(|f| CompiledFunction {
31            name: f.name.clone(),
32            frame_size: f.frame_size + f.reg_count, // locals + spilled registers
33            chunk: function_to_chunk(f),
34        })
35        .collect();
36    CompiledProgram {
37        script,
38        functions,
39        // The script has no Resolver locals (its variables are globals), but
40        // its spilled registers need a frame too.
41        script_frame_size: program.script.reg_count,
42    }
43}

45struct ChunkBuilder<'a> {
46    f: &'a IrFunction,
47    chunk: Chunk,
48    /// Bytecode offset where each block starts (filled as blocks are emitted).
49    block_starts: Vec<Option<usize>>,
50    /// (operand position, target block) pairs to patch once starts are known.
51    pending_jumps: Vec<(usize, BlockId)>,
52}
53
54fn function_to_chunk(f: &IrFunction) -> Chunk {
55    let mut b = ChunkBuilder {
56        f,
57        chunk: Chunk::new(),
58        block_starts: vec![None; f.blocks.len()],
59        pending_jumps: Vec::new(),
60    };
61
62    for (id, block) in f.blocks.iter().enumerate() {
63        if block.dead {
64            continue;
65        }
66        b.block_starts[id] = Some(b.chunk.code.len());
67        b.emit_block(block);
68    }
69
70    for (operand_at, target) in std::mem::take(&mut b.pending_jumps) {
71        let target_start = b.block_starts[target.0 as usize]
72            .expect("live blocks never jump to dead blocks");
73        debug_assert!(target_start >= operand_at + 2, "CFG edges are forward-only (DAG)");
74        b.chunk.patch_jump_to(operand_at, target_start);
75    }
76
77    b.chunk
78}
79
80impl ChunkBuilder<'_> {
81    /// The frame slot a virtual register is spilled to.
82    fn reg_slot(&self, reg: Reg) -> u16 {
83        u16::try_from(self.f.frame_size + reg.0).expect("spilled register slot exceeded u16::MAX")
84    }
85
86    fn slot_u16(slot: SlotIndex) -> u16 {
87        u16::try_from(slot.0).expect("slot index exceeded u16::MAX")
88    }
89
90    /// Push a register's value onto the operand stack.
91    fn push_reg(&mut self, reg: Reg, offset: u32) {
92        let slot = self.reg_slot(reg);
93        self.chunk.write_op(OpCode::GetLocal, offset);
94        self.chunk.write_u16(slot, offset);
95    }
96
97    /// Pop the top of the operand stack into a register.
98    fn pop_into_reg(&mut self, reg: Reg, offset: u32) {
99        let slot = self.reg_slot(reg);
100        self.chunk.write_op(OpCode::SetLocal, offset); // SetLocal peeks…
101        self.chunk.write_u16(slot, offset);
102        self.chunk.write_op(OpCode::Pop, offset); // …so pop the leftover
103    }
104
105    fn emit_jump_to(&mut self, op: OpCode, target: BlockId, offset: u32) {
106        let operand_at = self.chunk.emit_jump(op, offset);
107        self.pending_jumps.push((operand_at, target));
108    }
109
110    fn emit_block(&mut self, block: &Block) {
111        for instr in &block.instrs {
112            self.emit_instr(&instr.kind, instr.offset);
113        }
114        let offset = block.term_offset;
115        match &block.terminator {
116            Terminator::Jump(target) => self.emit_jump_to(OpCode::Jump, *target, offset),
117            Terminator::Branch { cond, then_block, else_block } => {
118                // JumpIfFalse peeks, so both paths pop the condition copy.
119                self.push_reg(*cond, offset);
120                let to_else = self.chunk.emit_jump(OpCode::JumpIfFalse, offset);
121                self.chunk.write_op(OpCode::Pop, offset);
122                self.emit_jump_to(OpCode::Jump, *then_block, offset);
123                let else_entry = self.chunk.code.len();
124                self.chunk.patch_jump_to(to_else, else_entry);
125                self.chunk.write_op(OpCode::Pop, offset);
126                self.emit_jump_to(OpCode::Jump, *else_block, offset);
127            }
128            Terminator::Return(reg) => {
129                self.push_reg(*reg, offset);
130                self.chunk.write_op(OpCode::Return, offset);
131            }
132        }
133    }
134
135    fn emit_instr(&mut self, kind: &InstrKind, offset: u32) {
136        use InstrKind::*;
137        match kind {
138            Const { dst, value } => {
139                let idx = self.chunk.add_constant(value.clone());
140                self.chunk.write_op(OpCode::Constant, offset);
141                self.chunk.write_u16(idx, offset);
142                self.pop_into_reg(*dst, offset);
143            }
144            Binary { dst, op, lhs, rhs, .. } => {
145                self.push_reg(*lhs, offset);
146                self.push_reg(*rhs, offset);
147                self.chunk.write_op(binary_opcode(op), offset);
148                self.pop_into_reg(*dst, offset);
149            }
150            Negate { dst, src, .. } => {
151                self.push_reg(*src, offset);
152                self.chunk.write_op(OpCode::Negate, offset);
153                self.pop_into_reg(*dst, offset);
154            }
155            Not { dst, src } => {
156                self.push_reg(*src, offset);
157                self.chunk.write_op(OpCode::Not, offset);
158                self.pop_into_reg(*dst, offset);
159            }
160            // to_bool == not(not(x)) — reuses the existing opcodes.
161            ToBool { dst, src } => {
162                self.push_reg(*src, offset);
163                self.chunk.write_op(OpCode::Not, offset);
164                self.chunk.write_op(OpCode::Not, offset);
165                self.pop_into_reg(*dst, offset);
166            }
167            ToFloat { dst, src } => {
168                self.push_reg(*src, offset);
169                self.chunk.write_op(OpCode::ToFloat, offset);
170                self.pop_into_reg(*dst, offset);
171            }
172            LoadLocal { dst, slot } => {
173                self.chunk.write_op(OpCode::GetLocal, offset);
174                self.chunk.write_u16(Self::slot_u16(*slot), offset);
175                self.pop_into_reg(*dst, offset);
176            }
177            StoreLocal { slot, src } => {
178                self.push_reg(*src, offset);
179                self.chunk.write_op(OpCode::SetLocal, offset);
180                self.chunk.write_u16(Self::slot_u16(*slot), offset);
181                self.chunk.write_op(OpCode::Pop, offset);
182            }
183            LoadGlobal { dst, slot } => {
184                self.chunk.write_op(OpCode::GetGlobal, offset);
185                self.chunk.write_u16(Self::slot_u16(*slot), offset);
186                self.pop_into_reg(*dst, offset);
187            }
188            StoreGlobal { slot, src } => {
189                self.push_reg(*src, offset);
190                self.chunk.write_op(OpCode::SetGlobal, offset);
191                self.chunk.write_u16(Self::slot_u16(*slot), offset);
192                self.chunk.write_op(OpCode::Pop, offset);
193            }
194            Call { dst, function, args } => {
195                for arg in args {
196                    self.push_reg(*arg, offset);
197                }
198                self.chunk.write_op(OpCode::Call, offset);
199                self.chunk.write_u16(u16::try_from(*function).expect("function id exceeded u16::MAX"), offset);
200                self.chunk.write_u8(u8::try_from(args.len()).expect("more than 255 arguments"), offset);
201                self.pop_into_reg(*dst, offset);
202            }
203            CallBuiltin { dst, builtin, args } => {
204                for arg in args {
205                    self.push_reg(*arg, offset);
206                }
207                let op = match builtin {
208                    BuiltinFn::Print => OpCode::Print,
209                    BuiltinFn::Max => OpCode::Max,
210                    BuiltinFn::Min => OpCode::Min,
211                };
212                self.chunk.write_op(op, offset);
213                self.chunk.write_u8(u8::try_from(args.len()).expect("more than 255 arguments"), offset);
214                self.pop_into_reg(*dst, offset);
215            }
216        }
217    }
218}

220fn binary_opcode(kind: &crate::ast::ASTBinaryOperatorKind) -> OpCode {
221    use crate::ast::ASTBinaryOperatorKind::*;
222    match kind {
223        Plus => OpCode::Add,
224        Minus => OpCode::Subtract,
225        Multiply => OpCode::Multiply,
226        Divide => OpCode::Divide,
227        Modulo => OpCode::Modulo,
228        Exponentiation => OpCode::Power,
229        BitwiseAnd => OpCode::BitAnd,
230        BitwiseOr => OpCode::BitOr,
231        BitwiseXor => OpCode::BitXor,
232        LeftShift => OpCode::ShiftLeft,
233        RightShift => OpCode::ShiftRight,
234        Equal => OpCode::Equal,
235        NotEqual => OpCode::NotEqual,
236        Less => OpCode::Less,
237        Greater => OpCode::Greater,
238        LessEqual => OpCode::LessEqual,
239        GreaterEqual => OpCode::GreaterEqual,
240        LogicalAnd | LogicalOr => unreachable!("logical operators lower to control flow"),
241    }
