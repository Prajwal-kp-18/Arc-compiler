1//! # Optimization Passes
2//!
3//! Constant folding, dead code elimination, and local common-subexpression
4//! elimination, run per-function to a fixpoint. Every pass obeys one law:
5//! **behavior is untouchable** — including *runtime errors*. A fold that
6//! would error doesn't fold (`1 / 0` stays and fails at runtime, same line,
7//! same message); DCE and CSE only touch instructions whose `can_error`
8//! fact (computed at lowering from the Resolver's static types) proves they
9//! cannot raise. The golden suite runs the whole pipeline under `--opt` and
10//! must stay byte-identical.
11

19/// Runs the full pass pipeline (fold → DCE → CSE, to fixpoint) on every
20/// function.
21pub fn optimize(program: &mut IrProgram) {
22    optimize_function(&mut program.script);
23    for f in &mut program.functions {
24        optimize_function(f);
25    }
26}
27
28fn optimize_function(f: &mut IrFunction) {
29    // Converges in one or two rounds in practice; the cap is a safety net
30    // against a pass-interaction cycle, not a tuning knob.
31    for _ in 0..10 {
32        let mut changed = fold(f);
33        changed |= dce(f);
34        changed |= cse(f);
35        if !changed {
36            return;
37        }
38    }
39}

41/// How many times each register is written across the whole function.
42/// Almost always 1; `&&`/`||` join results are written in two arm blocks
43/// and are excluded from const-tracking and CSE caching.
44fn write_counts(f: &IrFunction) -> HashMap<Reg, u32> {
45    let mut counts = HashMap::new();
46    for block in f.blocks.iter().filter(|b| !b.dead) {
47        for instr in &block.instrs {
48            if let Some(dst) = instr.kind.dst() {
49                *counts.entry(dst).or_insert(0) += 1;
50            }
51        }
52    }
53    counts
54}

56// -- constant folding ---------------------------------------------------------
57
58pub fn fold(f: &mut IrFunction) -> bool {
59    let counts = write_counts(f);
60    let mut consts: HashMap<Reg, Value> = HashMap::new();
61    let mut changed = false;
62
63    // Pre-seed with every existing Const instruction, regardless of block —
64    // a plain collection pass, order-independent, so it finds constants
65    // inside a loop body exactly as readily as ones before the loop.
66    for block in f.blocks.iter().filter(|b| !b.dead) {
67        for instr in &block.instrs {
68            if let InstrKind::Const { dst, value } = &instr.kind {
69                if counts.get(dst) == Some(&1) {
70                    consts.insert(*dst, value.clone());
71                }
72            }
73        }
74    }
75
76    for block in f.blocks.iter_mut().filter(|b| !b.dead) {
77        for instr in &mut block.instrs {
78            let folded: Option<(Reg, Value)> = match &instr.kind {
79                InstrKind::Binary { dst, op, lhs, rhs, .. } => {
80                    match (consts.get(lhs), consts.get(rhs)) {
81                        (Some(l), Some(r)) => fold_binary(*op, l, r).map(|v| (*dst, v)),
82                        _ => None,
83                    }
84                }
85                InstrKind::Negate { dst, src, .. } => match consts.get(src) {
86                    Some(Value::Integer(i)) => Some((*dst, Value::Integer(-i))),
87                    Some(Value::Float(fl)) => Some((*dst, Value::Float(-fl))),
88                    _ => None, // non-numeric const: leave the runtime error in place
89                },
90                InstrKind::Not { dst, src } => {
91                    consts.get(src).map(|v| (*dst, Value::Boolean(!v.to_boolean())))
92                }
93                InstrKind::ToBool { dst, src } => {
94                    consts.get(src).map(|v| (*dst, Value::Boolean(v.to_boolean())))
95                }
96                InstrKind::ToFloat { dst, src } => consts.get(src).map(|v| {
97                    let widened = match v {
98                        Value::Integer(i) => Value::Float(*i as f64),
99                        other => other.clone(), // runtime ToFloat passes non-Integers through
100                    };
101                    (*dst, widened)
102                }),
103                _ => None,
104            };
105
106            if let Some((dst, value)) = folded {
107                if counts.get(&dst) == Some(&1) {
108                    consts.insert(dst, value.clone());
109                }
110                instr.kind = InstrKind::Const { dst, value };
111                changed = true;
112            }
113        }
114
115        // A branch on a constant condition becomes a jump; the untaken
116        // block loses a predecessor and DCE's reachability sweep drops it.
117        if let Terminator::Branch { cond, then_block, else_block } = &block.terminator {
118            if let Some(v) = consts.get(cond) {
119                let target = if v.to_boolean() { *then_block } else { *else_block };
120                block.terminator = Terminator::Jump(target);
121                changed = true;
122            }
123        }
124    }
125    changed
126}

128/// Folds one binary op over constants via the same `apply_binary` both
129/// backends execute — a folded result can never differ from the runtime
130/// result. Returns `None` (no fold) where runtime behavior must be
131/// preserved: ops that would error, and `Int ** Int` cases where the
132/// runtime's unchecked `i64::pow` would panic (folding those would move the
133/// panic to compile time).
134fn fold_binary(op: ASTBinaryOperatorKind, l: &Value, r: &Value) -> Option<Value> {
135    if op == ASTBinaryOperatorKind::Exponentiation {
136        if let (Value::Integer(a), Value::Integer(b)) = (l, r) {
137            if *b >= 0 {
138                let exp = u32::try_from(*b).ok()?;
139                return a.checked_pow(exp).map(Value::Integer);
140            }
141            // negative exponent: falls through to apply_binary's Float path
142        }
143    }
144    apply_binary(&op, l, r).ok()
145}

147// -- dead code elimination ------------------------------------------------------
148
149pub fn dce(f: &mut IrFunction) -> bool {
150    let mut changed = mark_unreachable_blocks(f);
151
152    // Deleting an instruction can orphan its operands' definitions, so
153    // sweep to a local fixpoint.
154    loop {
155        let mut used: std::collections::HashSet<Reg> = std::collections::HashSet::new();
156        for block in f.blocks.iter_mut().filter(|b| !b.dead) {
157            for instr in &mut block.instrs {
158                instr.kind.for_each_use(&mut |r| {
159                    used.insert(*r);
160                });
161            }
162            match &block.terminator {
163                Terminator::Branch { cond, .. } => {
164                    used.insert(*cond);
165                }
166                Terminator::Return(r) => {
167                    used.insert(*r);
168                }
169                Terminator::Jump(_) => {}
170            }
171        }
172
173        let mut swept = false;
174        for block in f.blocks.iter_mut().filter(|b| !b.dead) {
175            let before = block.instrs.len();
176            block.instrs.retain(|instr| {
177                let dead = instr.kind.is_pure() && instr.kind.dst().is_some_and(|d| !used.contains(&d));
178                !dead
179            });
180            swept |= block.instrs.len() != before;
181        }
182        if !swept {
183            break;
184        }
185        changed = true;
186    }
187    changed
188}
189
190fn mark_unreachable_blocks(f: &mut IrFunction) -> bool {
191    let mut reachable = vec![false; f.blocks.len()];
192    let mut worklist = vec![0usize];
193    while let Some(id) = worklist.pop() {
194        if reachable[id] {
195            continue;
196        }
197        reachable[id] = true;
198        match &f.blocks[id].terminator {
199            Terminator::Jump(b) => worklist.push(b.0 as usize),
200            Terminator::Branch { then_block, else_block, .. } => {
201                worklist.push(then_block.0 as usize);
202                worklist.push(else_block.0 as usize);
203            }
204            Terminator::Return(_) => {}
205        }
206    }
207
208    let mut changed = false;
209    for (id, block) in f.blocks.iter_mut().enumerate() {
210        if !reachable[id] && !block.dead {
211            block.dead = true;
212            changed = true;
213        }
214    }
215    changed
216}

220/// Value-numbering key for a repeatable computation. `Const` is not CSE'd
221/// (`Value` isn't hashable; fold + DCE already collapse constant chains).
222#[derive(PartialEq, Eq, Hash)]
223enum Key {
224    Binary(ASTBinaryOperatorKind, Reg, Reg),
225    Negate(Reg),
226    Not(Reg),
227    ToBool(Reg),
228    ToFloat(Reg),
229    LoadLocal(u32),
230    LoadGlobal(u32),
231}
232
233pub fn cse(f: &mut IrFunction) -> bool {
234    let counts = write_counts(f);
235    // dup register -> the earlier register holding the same value. Filled
236    // per block (`available` below is local to each block: CSE itself is
237    // block-local), applied to operands on the fly as blocks are walked in
238    // id order. Safe even with `while`'s back-edge: every register still has
239    // exactly one static def site (the loop header's condition is one
240    // instruction, re-executed at runtime, not redefined per iteration), so
241    // substitutions never point at a not-yet-processed block.
242    let mut subst: HashMap<Reg, Reg> = HashMap::new();
243    let mut changed = false;
244
245    for block in f.blocks.iter_mut().filter(|b| !b.dead) {
246        let mut available: HashMap<Key, Reg> = HashMap::new();
247
248        block.instrs.retain_mut(|instr| {
249            // Canonicalize operands through earlier substitutions first so
250            // later duplicates match on canonical registers.
251            instr.kind.for_each_use(&mut |r| {
252                while let Some(rep) = subst.get(r) {
253                    *r = *rep;
254                }
255            });
256
257            let key = match &instr.kind {
258                // Only provably non-erroring ops are CSE-able: merging two
259                // *failing* ops would change how many errors the
260                // tree-walker reports (design doc §5.3).
261                InstrKind::Binary { op, lhs, rhs, can_error: false, .. } => Some(Key::Binary(*op, *lhs, *rhs)),
262                InstrKind::Negate { src, can_error: false, .. } => Some(Key::Negate(*src)),
263                InstrKind::Not { src, .. } => Some(Key::Not(*src)),
264                InstrKind::ToBool { src, .. } => Some(Key::ToBool(*src)),
265                InstrKind::ToFloat { src, .. } => Some(Key::ToFloat(*src)),
266                InstrKind::LoadLocal { slot, .. } => Some(Key::LoadLocal(slot.0)),
267                InstrKind::LoadGlobal { slot, .. } => Some(Key::LoadGlobal(slot.0)),
268                _ => None,
269            };
270
271            // Invalidate stale loads before (possibly) caching this instr.
272            match &instr.kind {
273                InstrKind::StoreLocal { slot, .. } => {
274                    available.remove(&Key::LoadLocal(slot.0));
275                }
276                InstrKind::StoreGlobal { slot, .. } => {
277                    available.remove(&Key::LoadGlobal(slot.0));
278                }
279                // Calls may write globals; one conservative rule beats
280                // three clever ones — drop every cached load.
281                InstrKind::Call { .. } | InstrKind::CallBuiltin { .. } => {
282                    available.retain(|k, _| !matches!(k, Key::LoadLocal(_) | Key::LoadGlobal(_)));
283                }
284                _ => {}
285            }
286
287            let (Some(key), Some(dst)) = (key, instr.kind.dst()) else {
288                return true;
289            };
290            // Multi-write registers (&&/|| results) are never cached or
291            // replaced — their value depends on which arm executed.
292            if counts.get(&dst) != Some(&1) {
293                return true;
294            }
295
296            if let Some(&earlier) = available.get(&key) {
297                subst.insert(dst, earlier);
298                changed = true;
299                false // drop the duplicate computation
300            } else {
301                available.insert(key, dst);
302                true
303            }
304        });
305
306        // The block's terminator may read a substituted register.
307        match &mut block.terminator {
308            Terminator::Branch { cond, .. } => canonicalize(cond, &subst),
309            Terminator::Return(r) => canonicalize(r, &subst),
310            Terminator::Jump(_) => {}
311        }
312    }
313
314    // Substitutions from one block can be used in later blocks (a dup in b1
315    // whose dst is read in b3) — rewrite everything once at the end too.
316    if changed {
317        for block in f.blocks.iter_mut().filter(|b| !b.dead) {
318            for instr in &mut block.instrs {
319                instr.kind.for_each_use(&mut |r| canonicalize_mut(r, &subst));
320            }
321            match &mut block.terminator {
322                Terminator::Branch { cond, .. } => canonicalize(cond, &subst),
323                Terminator::Return(r) => canonicalize(r, &subst),
324                Terminator::Jump(_) => {}
325            }
326        }
327    }
328    changed
329}
330
331fn canonicalize(r: &mut Reg, subst: &HashMap<Reg, Reg>) {
332    canonicalize_mut(r, subst);
333}
334
335fn canonicalize_mut(r: &mut Reg, subst: &HashMap<Reg, Reg>) {
336    while let Some(rep) = subst.get(r) {
337        *r = *rep;
338    }
