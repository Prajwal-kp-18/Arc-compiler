1//! # IR
2//!
3//! Phase 4 of the compiler pipeline: a three-address-code intermediate
4//! representation with basic blocks and explicit control flow, sitting
5//! between the resolved AST and codegen. [`lower`] builds it, [`passes`]
6//! optimizes it (constant folding, DCE, local CSE — all gated on preserving
7//! behavior *including runtime errors*), [`to_bytecode`] lowers it back to
8//! chunks so the existing VM executes it (`--opt`), and [`dump`] prints it
9//! (`--dump-ir`, `--dump-ir=opt`). See `local-notes/phase4-ir-design.md`.
10

12pub mod lower;
13pub mod liveness;
14pub mod passes;
15pub mod dump;
16pub mod to_bytecode;

31    fn lower_source(source: &str) -> IrProgram {
32        let mut lexer = Lexer::new(source);
33        let mut tokens = Vec::new();
34        while let Some(token) = lexer.next_token() {
35            tokens.push(token);
36        }
37        let mut parser = Parser::new(tokens);
38        let mut ast = Ast::new();
39        while let Some(stmt) = parser.next_statement() {
40            ast.add_statement(stmt);
41        }
42        assert!(parser.diagnostics.is_empty(), "parse errors: {:?}", parser.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>());
43
44        let mut resolver = Resolver::new();
45        resolver.resolve(&ast);
46        assert!(!resolver.has_errors(), "resolve errors: {:?}", resolver.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>());
47
48        lower::IrLowering::new(resolver.function_count()).lower(&ast)
49    }

51    /// Counts live instructions matching `pred` across the whole program.
52    fn count_instrs(program: &IrProgram, pred: impl Fn(&InstrKind) -> bool) -> usize {
53        std::iter::once(&program.script)
54            .chain(program.functions.iter())
55            .flat_map(|f| f.blocks.iter().filter(|b| !b.dead))
56            .flat_map(|b| b.instrs.iter())
57            .filter(|i| pred(&i.kind))
58            .count()
59    }

61    /// Runs a source program through tree-walker and the *optimized* IR
62    /// pipeline, asserting identical last_value and error messages.
63    fn assert_optimized_parity(source: &str) {
64        let mut lexer = Lexer::new(source);
65        let mut tokens = Vec::new();
66        while let Some(token) = lexer.next_token() {
67            tokens.push(token);
68        }
69        let mut parser = Parser::new(tokens);
70        let mut ast = Ast::new();
71        while let Some(stmt) = parser.next_statement() {
72            ast.add_statement(stmt);
73        }
74        let mut resolver = Resolver::new();
75        resolver.resolve(&ast);
76        assert!(!resolver.has_errors());
77
78        let mut evaluator = ASTEvaluator::new(resolver.global_slot_count());
79        evaluator.execute(&ast);
80
81        let mut program = lower::IrLowering::new(resolver.function_count()).lower(&ast);
82        passes::optimize(&mut program);
83        let compiled = to_bytecode::program_to_bytecode(&program);
84        let mut vm = VM::new(&compiled, resolver.global_slot_count());
85        vm.run();
86
87        assert_eq!(evaluator.last_value, vm.last_value, "last_value diverged on: {}", source);
88        let tree_errors: Vec<_> = evaluator.errors.iter().map(|d| d.message.clone()).collect();
89        let vm_errors: Vec<_> = vm.errors.iter().map(|d| d.message.clone()).collect();
90        assert_eq!(tree_errors, vm_errors, "errors diverged on: {}", source);
91    }

253    fn test_liveness_dead_store_has_empty_live_after_and_live_store_is_live_after() {
254        // `a`'s store is dead (never read again); `b`'s store is live (read by return).
255        let program = lower_source("fn f(n: Int) { let a = 1; let b = 2; return b; } f(1);");
256        let f = &program.functions[0];
257        let live = liveness::analyze(f);
258        let block = &f.blocks[0];
259        let live_after = liveness::live_after_each(block, &live.live_out[0]);
260
261        let store_slots: Vec<u32> = block
262            .instrs
263            .iter()
264            .filter_map(|i| match &i.kind {
265                InstrKind::StoreLocal { slot, .. } => Some(slot.0),
266                _ => None,
267            })
268            .collect();
269        assert_eq!(store_slots.len(), 2, "expected two stores (a, b)\n{}", dump::dump_program(&program));
270        let (a_slot, b_slot) = (store_slots[0], store_slots[1]);
271
272        let idx_of = |slot: u32| {
273            block.instrs.iter().position(|i| matches!(&i.kind, InstrKind::StoreLocal { slot: s, .. } if s.0 == slot)).unwrap()
274        };
275        assert!(!live_after[idx_of(a_slot)].contains(&a_slot), "\n{}", dump::dump_program(&program));
276        assert!(live_after[idx_of(b_slot)].contains(&b_slot), "\n{}", dump::dump_program(&program));
277    }

299    fn test_liveness_loop_carried_store_stays_live_via_back_edge() {
300        let program = lower_source("fn f(n: Int) { let i = 0; while i < n { i = i + 1; } return i; } f(3);");
301        let f = &program.functions[0];
302        let live = liveness::analyze(f);
303        let i_slot = f.blocks[0]
304            .instrs
305            .iter()
306            .find_map(|i| match &i.kind {
307                InstrKind::StoreLocal { slot, .. } => Some(slot.0),
308                _ => None,
309            })
310            .unwrap();
311        // The loop body's `i = i + 1` store (not b0's initializer) must
312        // still be live-after: the back-edge feeds the header's condition
313        // read back into this block during the fixpoint.
314        let (bid, idx) = f
315            .blocks
316            .iter()
317            .enumerate()
318            .skip(1)
319            .find_map(|(bid, b)| {
320                let idx = b
321                    .instrs
322                    .iter()
323                    .position(|ins| matches!(&ins.kind, InstrKind::StoreLocal { slot, .. } if slot.0 == i_slot))?;
324                Some((bid, idx))
325            })
326            .unwrap_or_else(|| panic!("expected a loop-body store to slot {}\n{}", i_slot, dump::dump_program(&program)));
327        let live_after = liveness::live_after_each(&f.blocks[bid], &live.live_out[bid]);
328        assert!(live_after[idx].contains(&i_slot), "\n{}", dump::dump_program(&program));
329    }

425    #[test]
426    fn test_dse_preserves_runtime_error_from_dead_stores_source_expr() {
427        // `x`'s store is dead (never read again), but `10 % n` can raise
428        // "Modulo by zero" at n=0 — DSE must not silently swallow that
429        // error by deleting the op along with the dead store.
430        //
431        // Compared against the *un-optimized* VM rather than the
432        // tree-walker: the tree-walker wraps any erroring store RHS (`let`
433        // initializer or assignment) in an extra diagnostic the bytecode/IR
434        // paths never emit — a pre-existing divergence unrelated to DSE,
435        // reproducible on plain `--backend=vm` with no IR pass involved.
436        // This test isolates DSE's actual contract: it must not change the
437        // optimized pipeline's own error behavior relative to unoptimized.
438        let source = "fn f(n: Int) { let x = 10 % n; return 1; } f(0);";
439        let mut lexer = Lexer::new(source);
440        let mut tokens = Vec::new();
441        while let Some(token) = lexer.next_token() {
442            tokens.push(token);
443        }
444        let mut parser = Parser::new(tokens);
445        let mut ast = Ast::new();
446        while let Some(stmt) = parser.next_statement() {
447            ast.add_statement(stmt);
448        }
449        let mut resolver = Resolver::new();
450        resolver.resolve(&ast);
451        assert!(!resolver.has_errors());
452
453        let unopt = lower::IrLowering::new(resolver.function_count()).lower(&ast);
454        let unopt_compiled = to_bytecode::program_to_bytecode(&unopt);
455        let mut unopt_vm = VM::new(&unopt_compiled, resolver.global_slot_count());
456        unopt_vm.run();
457
458        let mut opt = lower::IrLowering::new(resolver.function_count()).lower(&ast);
459        passes::optimize(&mut opt);
460        let opt_compiled = to_bytecode::program_to_bytecode(&opt);
461        let mut opt_vm = VM::new(&opt_compiled, resolver.global_slot_count());
462        opt_vm.run();
463
464        let unopt_errors: Vec<_> = unopt_vm.errors.iter().map(|d| d.message.clone()).collect();
465        let opt_errors: Vec<_> = opt_vm.errors.iter().map(|d| d.message.clone()).collect();
466        assert_eq!(unopt_errors, opt_errors);
467        assert!(!opt_errors.is_empty(), "expected the modulo-by-zero error to survive DSE");
468    }

504    #[test]
505    fn test_passes_reduce_instruction_count() {
506        let source = "let a = 2 * 3 + 4; print(1 + 2 * 3, 2 ** 10, (4 - 2) * (4 - 2));";
507        let before = lower_source(source);
508        let mut after = lower_source(source);
509        passes::optimize(&mut after);
510        assert!(
511            after.instr_count() < before.instr_count(),
512            "expected a reduction, got {} -> {}\nbefore:\n{}\nafter:\n{}",
513            before.instr_count(),
514            after.instr_count(),
515            dump::dump_program(&before),
