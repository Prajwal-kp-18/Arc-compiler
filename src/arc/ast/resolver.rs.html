1//! # Resolver
2//!
3//! Walks an AST once, after parsing and before evaluation, and statically
4//! resolves every variable and function reference:
5//!
6//! - Every `let`/`const`/parameter gets a stable [`SlotIndex`] within its
7//!   enclosing frame (function-frame-relative for locals, a single flat
8//!   space for globals) — recorded directly on the AST node via `Cell`
9//!   fields, so the evaluator never does a runtime, string-keyed lookup.
10//! - Every expression gets a statically inferred [`DataType`], using the
11//!   same widening/error rules the old runtime coercion used, computed
12//!   ahead of time instead of on every evaluation. Where a type genuinely
13//!   can't be pinned down (an untyped parameter with no usage-derived type,
14//!   or a return type that only depends on itself with no reachable base
15//!   case), it falls back to `DataType::Any` and the Resolver emits a
16//!   non-fatal warning — this can never turn a previously-valid program
17//!   into a new hard failure.
18//! - `fn` declarations are resolved with lexical scoping *and* hoisting:
19//!   within a block, every sibling function's name/signature is registered
20//!   before any of their bodies are resolved, so forward references and
21//!   mutual recursion keep working without a runtime name-lookup stack.
22//! - Immutable-assignment, type-mismatch, and arity checks move here from
23//!   the evaluator, using the exact same diagnostic text, so a previously
24//!   *runtime* error now surfaces before evaluation ever starts.
25//!
26//! See `local-notes/phase1-resolver-design.md` for the full design writeup,
27//! including the nested-closure and recursive-return-type decisions.
28

50/// Where a variable reference resolves to.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub enum ResolvedBinding {
53    Local(SlotIndex),
54    Global(SlotIndex),
55}
56
57impl ResolvedBinding {
58    pub fn slot(&self) -> SlotIndex {
59        match self {
60            ResolvedBinding::Local(s) | ResolvedBinding::Global(s) => *s,
61        }
62    }
63}

217/// One function's local slot space (or the single global space, for frame 0).
218///
219/// Slots are never reused across sibling blocks within a frame — every
220/// nested if/block just keeps advancing `next_slot` — matching the flat
221/// "stack window addressed by slot index" Phase 3's VM frames will want.
222struct Frame {
223    is_global: bool,
224    next_slot: u32,
225    scopes: Vec<BlockScope>,
226}

228/// Resolves an [`Ast`] in place, annotating it with slots/types/bindings.
229///
230/// Nested functions do **not** capture their enclosing function's locals
231/// (see the design doc, §9) — only their own frame and the single global
232/// frame are visible to a function body. Sibling function names *are*
233/// visible regardless of declaration order, via hoisting.
234pub struct Resolver {
235    frames: Vec<Frame>,
236    next_function_id: u32,
237    function_param_types: HashMap<u32, Vec<DataType>>,
238    function_names: HashMap<u32, String>,
239    function_return_types: HashMap<u32, DataType>,
240    /// FunctionIds currently being resolved (self/mutual recursion guard).
241    currently_resolving: Vec<u32>,
242    /// Stack of "types seen at `return` statements", one per function body
243    /// currently being resolved; empty means we're not inside any function.
244    return_types_stack: Vec<Vec<DataType>>,
245    pub diagnostics: Vec<Diagnostic>,
246}

248impl Resolver {

250        Self {
251            frames: vec![Frame { is_global: true, next_slot: 0, scopes: Vec::new() }],
252            next_function_id: 0,
253            function_param_types: HashMap::new(),
254            function_names: HashMap::new(),
255            function_return_types: HashMap::new(),
256            currently_resolving: Vec::new(),
257            return_types_stack: Vec::new(),
258            diagnostics: Vec::new(),
259        }

262    /// Resolves a program's statements against the persistent global scope.
263    ///
264    /// Safe to call repeatedly with new statements (e.g. once per REPL
265    /// line) — the global scope is never popped, so later calls still see
266    /// everything declared by earlier ones. Call [`Resolver::has_errors`]
267    /// afterward to decide whether it's safe to evaluate.
268    pub fn resolve(&mut self, ast: &Ast) {
269        if self.frames[0].scopes.is_empty() {
270            self.frames[0].scopes.push(BlockScope::new());
271        }
272        self.hoist_and_resolve(&ast.statements);
273    }
274
275    /// Number of global slots needed — the evaluator allocates a `Vec<Value>` this size.
276    pub fn global_slot_count(&self) -> u32 {
277        self.frames[0].next_slot
278    }
279
280    /// Total number of function declarations resolved so far — `FunctionId`s
281    /// are dense and 0-based, so this also sizes a `Vec` indexable by them
282    /// (e.g. the bytecode compiler's per-function `Chunk` table).
283    pub fn function_count(&self) -> u32 {
284        self.next_function_id
285    }

287    pub fn has_errors(&self) -> bool {
288        self.diagnostics.iter().any(|d| d.severity == Severity::Error)
289    }

313    fn resolve_statements(&mut self, stmts: &[ASTStatement]) {
314        self.frames.last_mut().unwrap().scopes.push(BlockScope::new());
315        self.hoist_and_resolve(stmts);
316        self.frames.last_mut().unwrap().scopes.pop();
317    }
318
319    /// Hoists every direct `fn` declaration in `stmts`, then resolves each
320    /// statement in order (function bodies included, at their own position).
321    fn hoist_and_resolve(&mut self, stmts: &[ASTStatement]) {
322        self.hoist_functions(stmts);
323        for stmt in stmts {
324            self.visit_statement(stmt);
325        }
326    }
327
328    fn hoist_functions(&mut self, stmts: &[ASTStatement]) {
329        for stmt in stmts {
330            if let ASTStatementKind::FunctionDeclaration(func) = &stmt.kind {
331                let id = FunctionId(self.next_function_id);
332                self.next_function_id += 1;
333                func.function_id.set(Some(id));
334
335                let param_types: Vec<DataType> = func
336                    .parameters
337                    .iter()
338                    .map(|p| p.type_annotation.unwrap_or(DataType::Any))
339                    .collect();
340
341                let frame = self.frames.last_mut().unwrap();
342                let scope = frame.scopes.last_mut().unwrap();
343                if scope.functions.contains_key(&func.name) {
344                    self.error(format!("Function '{}' already declared in this scope", func.name));
345                } else {
346                    scope.functions.insert(func.name.clone(), id);
347                }
348
349                self.function_param_types.insert(id.0, param_types);
350                self.function_names.insert(id.0, func.name.clone());
351            }
352        }
353    }

355    fn declare_variable(
356        &mut self,
357        name: &str,
358        data_type: DataType,
359        is_mutable: bool,
360        span: Option<TextSpan>,
361    ) -> ResolvedBinding {
362        let frame_idx = self.frames.len() - 1;
363        let is_global = self.frames[frame_idx].is_global;
364
365        if let Some(existing) = self.frames[frame_idx].scopes.last().unwrap().variables.get(name) {
366            let slot = existing.slot;
367            self.error_at(format!("Variable '{}' already declared in this scope", name), span);
368            return if is_global { ResolvedBinding::Global(slot) } else { ResolvedBinding::Local(slot) };
369        }
370
371        let slot = SlotIndex(self.frames[frame_idx].next_slot);
372        self.frames[frame_idx].next_slot += 1;
373        self.frames[frame_idx]
374            .scopes
375            .last_mut()
376            .unwrap()
377            .variables
378            .insert(name.to_string(), VarInfo { slot, data_type, is_mutable });
379
380        if is_global { ResolvedBinding::Global(slot) } else { ResolvedBinding::Local(slot) }
381    }
382
383    /// Searches the current frame's scope stack, then falls back to the
384    /// global frame only — nested functions never see an intermediate
385    /// enclosing function's locals (see design doc §9).
386    fn lookup_variable(&self, name: &str) -> Option<(ResolvedBinding, DataType, bool)> {
387        let frame = self.frames.last().unwrap();
388        for scope in frame.scopes.iter().rev() {
389            if let Some(info) = scope.variables.get(name) {
390                let binding = if frame.is_global {
391                    ResolvedBinding::Global(info.slot)
392                } else {
393                    ResolvedBinding::Local(info.slot)
394                };
395                return Some((binding, info.data_type, info.is_mutable));
396            }
397        }
398        if !frame.is_global {
399            for scope in self.frames[0].scopes.iter().rev() {
400                if let Some(info) = scope.variables.get(name) {
401                    return Some((ResolvedBinding::Global(info.slot), info.data_type, info.is_mutable));
402                }
403            }
404        }
405        None
406    }
407
408    fn lookup_function(&self, name: &str) -> Option<FunctionId> {
409        let frame = self.frames.last().unwrap();
410        for scope in frame.scopes.iter().rev() {
411            if let Some(&id) = scope.functions.get(name) {
412                return Some(id);
413            }
414        }
415        if !frame.is_global {
416            for scope in self.frames[0].scopes.iter().rev() {
417                if let Some(&id) = scope.functions.get(name) {
418                    return Some(id);
419                }
420            }
421        }
422        None
423    }
424
425    fn visible_variable_names(&self) -> Vec<String> {
426        let mut names = Vec::new();
427        let mut seen = std::collections::HashSet::new();
428        let frame = self.frames.last().unwrap();
429        for scope in frame.scopes.iter().rev() {
430            for name in scope.variables.keys() {
431                if seen.insert(name.clone()) {
432                    names.push(name.clone());
433                }
434            }
435        }
436        if !frame.is_global {
437            for scope in self.frames[0].scopes.iter().rev() {
438                for name in scope.variables.keys() {
439                    if seen.insert(name.clone()) {
440                        names.push(name.clone());
441                    }
442                }
443            }
444        }
445        names
446    }

450    /// Resolves `expr`, records its type on the node, and returns it.
451    fn visit_expression_typed(&mut self, expr: &ASTExpression) -> DataType {
452        let ty = self.resolve_expr_type(expr);
453        expr.resolved_type.set(Some(ty));
454        ty
455    }
456
457    fn resolve_expr_type(&mut self, expression: &ASTExpression) -> DataType {
458        match &expression.kind {
459            ASTExpressionKind::Number(n) => n.value.get_type(),
460            ASTExpressionKind::Binary(bin) => self.resolve_binary(bin),
461            ASTExpressionKind::Paranthesized(p) => self.visit_expression_typed(&p.expression),
462            ASTExpressionKind::Unary(u) => self.resolve_unary(u),
463            ASTExpressionKind::PostfixUnary(p) => self.resolve_postfix(p),
464            ASTExpressionKind::Identifier(ident) => self.resolve_identifier(ident),
465            ASTExpressionKind::FunctionCall(call) => self.resolve_call(call),
466        }
467    }
468
469    fn resolve_identifier(&mut self, ident: &ASTIdentifierExpression) -> DataType {
470        let span = ident.token.as_ref().map(|t| t.span.clone());
471        match self.lookup_variable(&ident.name) {
472            Some((binding, data_type, _)) => {
473                ident.binding.set(Some(binding));
474                data_type
475            }
476            None => {
477                let message = format!("Variable '{}' not found", ident.name);
478                match closest_name(&ident.name, &self.visible_variable_names()) {
479                    Some(s) => self.error_with_suggestion(message, span, format!("did you mean '{}' ?", s)),
480                    None => self.error_at(message, span),
481                }
482                DataType::Any
483            }
484        }
485    }

487    /// Mirrors `Value::coerce_to_common_type`'s numeric-only cases (no
488    /// string concatenation) — shared by every arithmetic operator except `+`.
489    fn numeric_result(left: DataType, right: DataType) -> Option<DataType> {
490        use DataType::*;
491        match (left, right) {
492            (Integer, Integer) => Some(Integer),
493            (Float, Float) | (Integer, Float) | (Float, Integer) => Some(Float),
494            _ => None,
495        }
496    }
497
498    fn check_bitwise(&mut self, left: DataType, right: DataType, span: Option<TextSpan>, message: &str) -> DataType {
499        if left == DataType::Any || right == DataType::Any {
500            return DataType::Any;
501        }
502        // Value::to_integer() at runtime only fails for String.
503        if left == DataType::String || right == DataType::String {
504            self.error_at(message.to_string(), span);
505            return DataType::Any;
506        }
507        DataType::Integer
508    }
509
510    fn is_comparable_pair(left: DataType, right: DataType) -> bool {
511        left == right || matches!((left, right), (DataType::Integer, DataType::Float) | (DataType::Float, DataType::Integer))
512    }
513
514    fn check_equality(&mut self, left: DataType, right: DataType, span: Option<TextSpan>) -> DataType {
515        if left == DataType::Any || right == DataType::Any {
516            return DataType::Boolean;
517        }
518        if Self::is_comparable_pair(left, right) {
519            DataType::Boolean
520        } else {
521            self.error_at(format!("Cannot compare {:?} and {:?} for equality", left, right), span);
522            DataType::Any
523        }
524    }
525
526    fn check_ordering(&mut self, left: DataType, right: DataType, span: Option<TextSpan>) -> DataType {
527        if left == DataType::Any || right == DataType::Any {
528            return DataType::Boolean;
529        }
530        if Self::is_comparable_pair(left, right) {
531            DataType::Boolean
532        } else {
533            self.error_at(format!("Cannot compare {:?} and {:?}", left, right), span);
534            DataType::Any
535        }
536    }

538    fn resolve_binary(&mut self, bin: &ASTBinaryExpression) -> DataType {
539        let left = self.visit_expression_typed(&bin.left);
540        let right = self.visit_expression_typed(&bin.right);
541        let span = Some(bin.operator.token.span.clone());
542        use ASTBinaryOperatorKind::*;
543        use DataType::Any;
544
545        match bin.operator.kind {
546            // to_boolean() never fails, regardless of operand type.
547            LogicalAnd | LogicalOr => DataType::Boolean,
548
549            Plus => {
550                if left == Any || right == Any {
551                    Any
552                } else if left == DataType::String && right == DataType::String {
553                    DataType::String
554                } else if let Some(t) = Self::numeric_result(left, right) {
555                    t
556                } else {
557                    self.error_at(format!("Cannot add {:?} and {:?}", left, right), span);
558                    Any
559                }
560            }
561            Minus => {
562                if left == Any || right == Any {
563                    Any
564                } else if let Some(t) = Self::numeric_result(left, right) {
565                    t
566                } else {
567                    self.error_at(format!("Cannot subtract {:?} from {:?}", right, left), span);
568                    Any
569                }
570            }
571            Multiply => {
572                if left == Any || right == Any {
573                    Any
574                } else if let Some(t) = Self::numeric_result(left, right) {
575                    t
576                } else {
577                    self.error_at(format!("Cannot multiply {:?} and {:?}", left, right), span);
578                    Any
579                }
580            }
581            Divide => {
582                if left == Any || right == Any {
583                    Any
584                } else if let Some(t) = Self::numeric_result(left, right) {
585                    t
586                } else {
587                    self.error_at(format!("Cannot divide {:?} by {:?}", left, right), span);
588                    Any
589                }
590            }
591            Modulo => {
592                if left == Any || right == Any {
593                    Any
594                } else if let Some(t) = Self::numeric_result(left, right) {
595                    t
596                } else {
597                    self.error_at(format!("Cannot compute modulo of {:?} and {:?}", left, right), span);
598                    Any
599                }
600            }
601            Exponentiation => {
602                if left == Any || right == Any {
603                    Any
604                } else if let Some(t) = Self::numeric_result(left, right) {
605                    t
606                } else {
607                    self.error_at(format!("Cannot exponentiate {:?} and {:?}", left, right), span);
608                    Any
609                }
610            }
611            BitwiseAnd => self.check_bitwise(left, right, span, "Bitwise AND requires integer operands"),
612            BitwiseOr => self.check_bitwise(left, right, span, "Bitwise OR requires integer operands"),
613            BitwiseXor => self.check_bitwise(left, right, span, "Bitwise XOR requires integer operands"),
614            LeftShift => self.check_bitwise(left, right, span, "Left shift requires integer operands"),
615            RightShift => self.check_bitwise(left, right, span, "Right shift requires integer operands"),
616            Equal | NotEqual => self.check_equality(left, right, span),
617            Less | Greater | LessEqual | GreaterEqual => self.check_ordering(left, right, span),
618        }
619    }

621    fn resolve_unary(&mut self, u: &ASTUnaryExpression) -> DataType {
622        match u.operator.kind {
623            ASTUnaryOperatorKind::Increment | ASTUnaryOperatorKind::Decrement => {
624                self.resolve_inc_dec(&u.operand, matches!(u.operator.kind, ASTUnaryOperatorKind::Increment), false)
625            }
626            ASTUnaryOperatorKind::Plus => {
627                // Runtime passthrough: valid (and returns the operand type)
628                // for every type, never errors.
629                self.visit_expression_typed(&u.operand)
630            }
631            ASTUnaryOperatorKind::Minus => {
632                let ty = self.visit_expression_typed(&u.operand);
633                match ty {
634                    DataType::Any | DataType::Integer | DataType::Float => ty,
635                    other => {
636                        self.error(format!("Cannot negate {:?}", other));
637                        DataType::Any
638                    }
639                }
640            }
641            ASTUnaryOperatorKind::LogicalNot => {
642                self.visit_expression_typed(&u.operand);
643                DataType::Boolean
644            }
645        }
646    }

648    fn resolve_postfix(&mut self, p: &ASTPostfixUnaryExpression) -> DataType {
649        self.resolve_inc_dec(&p.operand, matches!(p.operator.kind, ASTUnaryOperatorKind::Increment), true)
650    }

652    fn resolve_inc_dec(&mut self, operand: &ASTExpression, is_increment: bool, is_postfix: bool) -> DataType {
653        let ident = match &operand.kind {
654            ASTExpressionKind::Identifier(ident) => ident,
655            _ => {
656                let prefix = if is_postfix { "Postfix " } else { "" };
657                self.error(format!("{}Increment/Decrement can only be applied to variables", prefix));
658                return DataType::Any;
659            }
660        };
661
662        if let Some((_, _, is_mutable)) = self.lookup_variable(&ident.name) {
663            if !is_mutable {
664                self.error(format!("Cannot assign to immutable variable '{}'", ident.name));
665            }
666        }
667
668        let ty = self.visit_expression_typed(operand);
669        match ty {
670            DataType::Any | DataType::Integer | DataType::Float => ty,
671            other => {
672                let verb = if is_increment { "increment" } else { "decrement" };
673                self.error(format!("Cannot {} {:?}", verb, other));
674                DataType::Any
675            }
676        }
677    }
678
679    fn resolve_call(&mut self, call: &ASTFunctionCallExpression) -> DataType {
680        let span = call.token.as_ref().map(|t| t.span.clone());
681        let arg_types: Vec<DataType> = call.arguments.iter().map(|a| self.visit_expression_typed(a)).collect();
682
683        if let Some(builtin) = builtin_named(&call.name) {
684            call.resolved_call.set(Some(ResolvedCall::Builtin(builtin)));
685            return self.resolve_builtin_call(builtin, &arg_types, span);
686        }
687
688        match self.lookup_function(&call.name) {
689            Some(id) => {
690                call.resolved_call.set(Some(ResolvedCall::User(id)));
691                let expected = self.function_param_types.get(&id.0).cloned().unwrap_or_default();
692                let name = self.function_names.get(&id.0).cloned().unwrap_or_else(|| call.name.clone());
693                if arg_types.len() != expected.len() {
694                    self.error_at(
695                        format!("Function '{}' expected {} arguments but got {}", name, expected.len(), arg_types.len()),
696                        span,
697                    );
698                    return DataType::Any;
699                }
700                for (i, (&expected_ty, &actual_ty)) in expected.iter().zip(arg_types.iter()).enumerate() {
701                    let widening_ok = expected_ty == DataType::Float && actual_ty == DataType::Integer;
702                    if expected_ty != DataType::Any && actual_ty != DataType::Any && expected_ty != actual_ty && !widening_ok {
703                        self.error_at(
704                            format!(
705                                "Type mismatch: argument {} to '{}' expected {:?} but got {:?}",
706                                i + 1, name, expected_ty, actual_ty
707                            ),
708                            span.clone(),
709                        );
710                    }
711                }
712                if self.currently_resolving.contains(&id.0) {
713                    // Direct or mutual recursion: this function's return
714                    // type isn't known yet.
715                    return DataType::Any;
716                }
717                self.function_return_types.get(&id.0).copied().unwrap_or(DataType::Any)
718            }
719            None => {
720                let mut names: Vec<String> = BUILTIN_NAMES.iter().map(|(n, _)| n.to_string()).collect();
721                names.extend(self.function_names.values().cloned());
722                let message = format!("Unknown function: '{}'", call.name);
723                match closest_name(&call.name, &names) {
724                    Some(s) => self.error_with_suggestion(message, span, format!("did you mean '{}' ?", s)),
725                    None => self.error_at(message, span),
726                }
727                DataType::Any
728            }
729        }
730    }

732    fn resolve_builtin_call(&mut self, builtin: BuiltinFn, arg_types: &[DataType], span: Option<TextSpan>) -> DataType {
733        use BuiltinFn::*;
734
735        // Fixed-arity builtins: (name, exact arg count, return type).
736        // Checked first so the per-builtin match below never sees a wrong
737        // argument count.
738        let fixed_arity: Option<usize> = match builtin {
739            Print | Max | Min => None, // variable-arity, checked separately below
740            Int | Float | Str | Len | Abs | Sqrt | Floor | Ceil | Round | Upper | Lower | Trim | Ord | Chr => Some(1),
741            Find | CharAt => Some(2),
742            Substr => Some(3),
743            Input | Clock => Some(0),
744            Assert => None, // 1 or 2, checked separately below
745        };
746        if let Some(expected) = fixed_arity {
747            if arg_types.len() != expected {
748                self.error_at(
749                    format!("{}() expects {} argument{} but got {}", builtin.name(), expected, if expected == 1 { "" } else { "s" }, arg_types.len()),
750                    span,
751                );
752                return DataType::Any;
753            }
754        }
755
756        match builtin {
757            // print() never produces a usable value.
758            Print => DataType::Any,
759            Max | Min => {
760                if arg_types.is_empty() {
761                    self.error_at(format!("{}() requires at least one argument", builtin.name()), span);
762                    return DataType::Any;
763                }
764                // ponytail: mismatched-argument-type errors inside max/min
765                // stay a runtime concern (Value::compare's error text);
766                // statically we just fall back to Any for a mixed call.
767                Self::combine_types(arg_types)
768            }
769            Int => DataType::Integer,
770            Float => DataType::Float,
771            Str => DataType::String,
772            Len => {
773                self.expect_type(arg_types[0], DataType::String, builtin.name(), span);
774                DataType::Integer
775            }
776            Input => DataType::String,
777            Substr => {
778                self.expect_type(arg_types[0], DataType::String, builtin.name(), span.clone());
779                self.expect_numeric(arg_types[1], builtin.name(), span.clone());
780                self.expect_numeric(arg_types[2], builtin.name(), span);
781                DataType::String
782            }
783            Find => {
784                self.expect_type(arg_types[0], DataType::String, builtin.name(), span.clone());
785                self.expect_type(arg_types[1], DataType::String, builtin.name(), span);
786                DataType::Integer
787            }
788            Upper | Lower | Trim | Ord => {
789                self.expect_type(arg_types[0], DataType::String, builtin.name(), span);
790                if builtin == Ord { DataType::Integer } else { DataType::String }
791            }
792            CharAt => {
793                self.expect_type(arg_types[0], DataType::String, builtin.name(), span.clone());
794                self.expect_numeric(arg_types[1], builtin.name(), span);
795                DataType::String
796            }
797            Chr => {
798                self.expect_numeric(arg_types[0], builtin.name(), span);
799                DataType::String
800            }
801            Abs => match arg_types[0] {
802                DataType::Integer => DataType::Integer,
803                DataType::Float => DataType::Float,
804                DataType::Any => DataType::Any,
805                other => {
806                    self.error_at(format!("abs() requires a numeric argument, got {:?}", other), span);
807                    DataType::Any
808                }
809            },
810            Sqrt => {
811                self.expect_numeric(arg_types[0], builtin.name(), span);
812                DataType::Float
813            }
814            Floor | Ceil | Round => {
815                self.expect_numeric(arg_types[0], builtin.name(), span);
816                DataType::Integer
817            }
818            Assert => {
819                if arg_types.is_empty() || arg_types.len() > 2 {
820                    self.error_at(format!("assert() expects 1 or 2 arguments but got {}", arg_types.len()), span);
821                }
822                DataType::Any
823            }
824            Clock => DataType::Float,
825        }
826    }
827
828    /// Emits an error unless `actual` is `expected` or statically unknown (`Any`).
829    fn expect_type(&mut self, actual: DataType, expected: DataType, who: &str, span: Option<TextSpan>) {
830        if actual != expected && actual != DataType::Any {
831            self.error_at(format!("{}() requires a {:?} argument, got {:?}", who, expected, actual), span);
832        }
833    }
834
835    /// Emits an error unless `actual` is `Integer`, `Float`, or statically unknown (`Any`).
836    fn expect_numeric(&mut self, actual: DataType, who: &str, span: Option<TextSpan>) {
837        if !matches!(actual, DataType::Integer | DataType::Float | DataType::Any) {
838            self.error_at(format!("{}() requires a numeric argument, got {:?}", who, actual), span);
839        }
840    }
841
842    /// Unifies a set of types the same way assignment widening does
843    /// (Integer+Float -> Float); anything else collapses to `Any`.
844    fn combine_types(types: &[DataType]) -> DataType {
845        let mut result = match types.first() {
846            Some(t) => *t,
847            None => return DataType::Any,
848        };
849        for &t in &types[1..] {
850            result = match (result, t) {
851                (a, b) if a == b => a,
852                (DataType::Integer, DataType::Float) | (DataType::Float, DataType::Integer) => DataType::Float,
853                _ => return DataType::Any,
854            };
855        }
856        result
857    }

861    fn resolve_function_body(&mut self, func: &ASTFunctionDeclaration, id: FunctionId, emit_warnings: bool) -> DataType {
862        self.frames.push(Frame { is_global: false, next_slot: 0, scopes: vec![BlockScope::new()] });
863
864        let mut param_types = Vec::new();
865        let mut any_untyped = false;
866        for param in &func.parameters {
867            let ty = match param.type_annotation {
868                Some(t) => t,
869                None => {
870                    any_untyped = true;
871                    DataType::Any
872                }
873            };
874            let binding = self.declare_variable(&param.name, ty, true, None);
875            param.slot.set(Some(binding.slot()));
876            param_types.push(ty);
877        }
878        self.function_param_types.insert(id.0, param_types);
879        self.function_names.insert(id.0, func.name.clone());
880
881        if any_untyped && emit_warnings {
882            self.warn(format!(
883                "cannot statically infer a type for one or more parameters of '{}'; falling back to dynamic type checking",
884                func.name
885            ));
886        }
887
888        // Params and the body's own top-level declarations share one scope
889        // (matching the old evaluator: shadowing a parameter at the body's
890        // top level was already a "redeclared in this scope" error).
891        self.return_types_stack.push(Vec::new());
892        self.hoist_and_resolve(&func.body);
893        let seen_returns = self.return_types_stack.pop().unwrap();
894
895        let return_type = if !seen_returns.is_empty() {
896            Self::combine_types(&seen_returns)
897        } else {
898            Self::last_expression_type(&func.body).unwrap_or(DataType::Any)
899        };
900
901        let frame_size = self.frames.last().unwrap().next_slot;
902        func.frame_size.set(Some(frame_size));
903        self.frames.pop();
904
905        return_type
906    }
907
908    fn last_expression_type(body: &[ASTStatement]) -> Option<DataType> {
909        match &body.last()?.kind {
910            ASTStatementKind::Expression(e) => e.resolved_type.get(),
911            _ => None,
912        }
913    }
914
915    fn calls_function_by_name(statements: &[ASTStatement], name: &str) -> bool {
916        statements.iter().any(|s| Self::stmt_calls(s, name))
917    }
918
919    fn stmt_calls(stmt: &ASTStatement, name: &str) -> bool {
920        match &stmt.kind {
921            ASTStatementKind::Expression(e) => Self::expr_calls(e, name),
922            ASTStatementKind::VariableDeclaration(d) => Self::expr_calls(&d.initializer, name),
923            ASTStatementKind::Assignment(a) => Self::expr_calls(&a.value, name),
924            ASTStatementKind::Block(stmts) => Self::calls_function_by_name(stmts, name),
925            ASTStatementKind::IfStatement(i) => {
926                Self::expr_calls(&i.condition, name)
927                    || Self::calls_function_by_name(&i.then_branch, name)
928                    || i.else_branch.as_ref().is_some_and(|b| Self::calls_function_by_name(b, name))
929            }
930            ASTStatementKind::WhileStatement(w) => {
931                Self::expr_calls(&w.condition, name) || Self::calls_function_by_name(&w.body, name)
932            }
933            // A nested fn's body is a separate scope; it doesn't count as
934            // the enclosing function calling itself.
935            ASTStatementKind::FunctionDeclaration(_) => false,
936            ASTStatementKind::ReturnStatement(r) => Self::expr_calls(&r.value, name),
937        }
938    }
939
940    fn expr_calls(expr: &ASTExpression, name: &str) -> bool {
941        match &expr.kind {
942            ASTExpressionKind::Number(_) | ASTExpressionKind::Identifier(_) => false,
943            ASTExpressionKind::Binary(b) => Self::expr_calls(&b.left, name) || Self::expr_calls(&b.right, name),
944            ASTExpressionKind::Paranthesized(p) => Self::expr_calls(&p.expression, name),
945            ASTExpressionKind::Unary(u) => Self::expr_calls(&u.operand, name),
946            ASTExpressionKind::PostfixUnary(p) => Self::expr_calls(&p.operand, name),
947            ASTExpressionKind::FunctionCall(c) => c.name == name || c.arguments.iter().any(|a| Self::expr_calls(a, name)),
948        }
949    }
950}

952impl ASTVisitor for Resolver {
953    // Bypassed entirely by our own `visit_expression` override below; kept
954    // only to satisfy the trait (it has no default implementation).
955    fn visit_number(&mut self, _number: &crate::ast::ASTNumberExpression) {}
956
957    fn visit_expression(&mut self, expression: &ASTExpression) {
958        self.visit_expression_typed(expression);
959    }
960
961    fn visit_variable_declaration(&mut self, decl: &ASTVariableDeclaration) {
962        let init_ty = self.visit_expression_typed(&decl.initializer);
963        let span = decl.name_span.as_ref().map(|t| t.span.clone());
964        let binding = self.declare_variable(&decl.name, init_ty, decl.is_mutable, span);
965        decl.binding.set(Some(binding));
966    }
967
968    fn visit_assignment(&mut self, assign: &ASTAssignment) {
969        let value_ty = self.visit_expression_typed(&assign.value);
970        let span = assign.name_span.as_ref().map(|t| t.span.clone());
971
972        match self.lookup_variable(&assign.name) {
973            None => {
974                let message = format!("Variable '{}' not found", assign.name);
975                match closest_name(&assign.name, &self.visible_variable_names()) {
976                    Some(s) => self.error_with_suggestion(message, span, format!("did you mean '{}' ?", s)),
977                    None => self.error_at(message, span),
978                }
979            }
980            Some((binding, declared_ty, is_mutable)) => {
981                assign.binding.set(Some(binding));
982                if !is_mutable {
983                    self.error_at(format!("Cannot assign to immutable variable '{}'", assign.name), span);
984                } else if value_ty != DataType::Any && declared_ty != DataType::Any && value_ty != declared_ty {
985                    let widening_ok = declared_ty == DataType::Float && value_ty == DataType::Integer;
986                    if widening_ok {
987                        assign.needs_float_widen.set(true);
988                    } else {
989                        self.error_at(
990                            format!(
991                                "Type mismatch: variable '{}' has type {:?}, cannot assign {:?}",
992                                assign.name, declared_ty, value_ty
993                            ),
994                            span,
995                        );
996                    }
997                }
998            }
999        }
1000    }
1001
1002    fn visit_function_declaration(&mut self, func: &ASTFunctionDeclaration) {
1003        let id = func.function_id.get().expect("function must be hoisted before its body is resolved");
1004
1005        self.currently_resolving.push(id.0);
1006        let tentative = self.resolve_function_body(func, id, true);
1007        self.currently_resolving.pop();
1008        self.function_return_types.insert(id.0, tentative);
1009
1010        let final_return = if tentative != DataType::Any && Self::calls_function_by_name(&func.body, &func.name) {
1011            let refined = self.resolve_function_body(func, id, false);
1012            self.function_return_types.insert(id.0, refined);
1013            refined
1014        } else {
1015            tentative
1016        };
1017
1018        if final_return == DataType::Any {
1019            self.warn(format!(
1020                "cannot statically infer a return type for '{}'; falling back to dynamic type checking",
1021                func.name
1022            ));
1023        }
1024
1025        func.inferred_return_type.set(Some(final_return));
1026    }
1027
1028    fn visit_return_statement(&mut self, ret: &ASTReturnStatement) {
1029        let ty = self.visit_expression_typed(&ret.value);
1030        match self.return_types_stack.last_mut() {
1031            Some(returns) => returns.push(ty),
1032            None => self.error("'return' can only be used inside a function".to_string()),
1033        }
1034    }
1035
1036    fn visit_if_statement(&mut self, if_stmt: &ASTIfStatement) {
1037        self.visit_expression_typed(&if_stmt.condition);
1038        self.resolve_statements(&if_stmt.then_branch);
1039        if let Some(else_branch) = &if_stmt.else_branch {
1040            self.resolve_statements(else_branch);
1041        }
1042    }
1043
1044    fn visit_block(&mut self, statements: &Vec<ASTStatement>) {
1045        self.resolve_statements(statements);
1046    }
1047
1048    fn visit_while_statement(&mut self, while_stmt: &ASTWhileStatement) {
1049        self.visit_expression_typed(&while_stmt.condition);
1050        self.resolve_statements(&while_stmt.body);
1051    }
1052}

1060    fn resolve(source: &str) -> (Ast, Resolver) {
1061        let mut lexer = Lexer::new(source);
1062        let mut tokens = Vec::new();
1063        while let Some(token) = lexer.next_token() {
1064            tokens.push(token);
1065        }
1066        let mut parser = Parser::new(tokens);
1067        let mut ast = Ast::new();
1068        while let Some(stmt) = parser.next_statement() {
1069            ast.add_statement(stmt);
1070        }
1071        assert!(parser.diagnostics.is_empty());
1072
1073        let mut resolver = Resolver::new();
1074        resolver.resolve(&ast);
1075        (ast, resolver)
1076    }
