1//! # Resolver
2//!
3//! Walks an AST once, after parsing and before evaluation, and statically
4//! resolves every variable and function reference:
5//!
6//! - Every `let`/`const`/parameter gets a stable [`SlotIndex`] within its
7//!   enclosing frame (function-frame-relative for locals, a single flat
8//!   space for globals) — recorded directly on the AST node via `Cell`
9//!   fields, so the evaluator never does a runtime, string-keyed lookup.
10//! - Every expression gets a statically inferred [`DataType`], using the
11//!   same widening/error rules the old runtime coercion used, computed
12//!   ahead of time instead of on every evaluation. Where a type genuinely
13//!   can't be pinned down (an untyped parameter with no usage-derived type,
14//!   or a return type that only depends on itself with no reachable base
15//!   case), it falls back to `DataType::Any` and the Resolver emits a
16//!   non-fatal warning — this can never turn a previously-valid program
17//!   into a new hard failure.
18//! - `fn` declarations are resolved with lexical scoping *and* hoisting:
19//!   within a block, every sibling function's name/signature is registered
20//!   before any of their bodies are resolved, so forward references and
21//!   mutual recursion keep working without a runtime name-lookup stack.
22//! - Immutable-assignment, type-mismatch, and arity checks move here from
23//!   the evaluator, using the exact same diagnostic text, so a previously
24//!   *runtime* error now surfaces before evaluation ever starts.
25//!
26//! See `local-notes/phase1-resolver-design.md` for the full design writeup,
27//! including the nested-closure and recursive-return-type decisions.
28

41/// A slot index within a frame (locals) or the global slot space.
42#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
43pub struct SlotIndex(pub u32);
44
45/// Stable identity for a resolved function declaration, assigned once by
46/// the Resolver and used by call sites instead of a runtime name lookup.
47#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
48pub struct FunctionId(pub u32);

50/// Where a variable reference resolves to.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub enum ResolvedBinding {
53    Local(SlotIndex),
54    Global(SlotIndex),
55}
56
57impl ResolvedBinding {
58    pub fn slot(&self) -> SlotIndex {
59        match self {
60            ResolvedBinding::Local(s) | ResolvedBinding::Global(s) => *s,
61        }
62    }
63}

107/// One function's local slot space (or the single global space, for frame 0).
108///
109/// Slots are never reused across sibling blocks within a frame — every
110/// nested if/block just keeps advancing `next_slot` — matching the flat
111/// "stack window addressed by slot index" Phase 3's VM frames will want.
112struct Frame {
113    is_global: bool,
114    next_slot: u32,
115    scopes: Vec<BlockScope>,
116}

118/// Resolves an [`Ast`] in place, annotating it with slots/types/bindings.
119///
120/// Nested functions do **not** capture their enclosing function's locals
121/// (see the design doc, §9) — only their own frame and the single global
122/// frame are visible to a function body. Sibling function names *are*
123/// visible regardless of declaration order, via hoisting.
124pub struct Resolver {
125    frames: Vec<Frame>,
126    next_function_id: u32,
127    function_param_types: HashMap<u32, Vec<DataType>>,
128    function_names: HashMap<u32, String>,
129    function_return_types: HashMap<u32, DataType>,
130    /// FunctionIds currently being resolved (self/mutual recursion guard).
131    currently_resolving: Vec<u32>,
132    /// Stack of "types seen at `return` statements", one per function body
133    /// currently being resolved; empty means we're not inside any function.
134    return_types_stack: Vec<Vec<DataType>>,
135    pub diagnostics: Vec<Diagnostic>,
136}

138impl Resolver {

140        Self {
141            frames: vec![Frame { is_global: true, next_slot: 0, scopes: Vec::new() }],
142            next_function_id: 0,
143            function_param_types: HashMap::new(),
144            function_names: HashMap::new(),
145            function_return_types: HashMap::new(),
146            currently_resolving: Vec::new(),
147            return_types_stack: Vec::new(),
148            diagnostics: Vec::new(),
149        }

152    /// Resolves a program's statements against the persistent global scope.
153    ///
154    /// Safe to call repeatedly with new statements (e.g. once per REPL
155    /// line) — the global scope is never popped, so later calls still see
156    /// everything declared by earlier ones. Call [`Resolver::has_errors`]
157    /// afterward to decide whether it's safe to evaluate.
158    pub fn resolve(&mut self, ast: &Ast) {
159        if self.frames[0].scopes.is_empty() {
160            self.frames[0].scopes.push(BlockScope::new());
161        }
162        self.hoist_and_resolve(&ast.statements);
163    }
164
165    /// Number of global slots needed — the evaluator allocates a `Vec<Value>` this size.
166    pub fn global_slot_count(&self) -> u32 {
167        self.frames[0].next_slot
168    }
169
170    pub fn has_errors(&self) -> bool {
171        self.diagnostics.iter().any(|d| d.severity == Severity::Error)
172    }

196    fn resolve_statements(&mut self, stmts: &[ASTStatement]) {
197        self.frames.last_mut().unwrap().scopes.push(BlockScope::new());
198        self.hoist_and_resolve(stmts);
199        self.frames.last_mut().unwrap().scopes.pop();
200    }
201
202    /// Hoists every direct `fn` declaration in `stmts`, then resolves each
203    /// statement in order (function bodies included, at their own position).
204    fn hoist_and_resolve(&mut self, stmts: &[ASTStatement]) {
205        self.hoist_functions(stmts);
206        for stmt in stmts {
207            self.visit_statement(stmt);
208        }
209    }
210
211    fn hoist_functions(&mut self, stmts: &[ASTStatement]) {
212        for stmt in stmts {
213            if let ASTStatementKind::FunctionDeclaration(func) = &stmt.kind {
214                let id = FunctionId(self.next_function_id);
215                self.next_function_id += 1;
216                func.function_id.set(Some(id));
217
218                let param_types: Vec<DataType> = func
219                    .parameters
220                    .iter()
221                    .map(|p| p.type_annotation.unwrap_or(DataType::Any))
222                    .collect();
223
224                let frame = self.frames.last_mut().unwrap();
225                let scope = frame.scopes.last_mut().unwrap();
226                if scope.functions.contains_key(&func.name) {
227                    self.error(format!("Function '{}' already declared in this scope", func.name));
228                } else {
229                    scope.functions.insert(func.name.clone(), id);
230                }
231
232                self.function_param_types.insert(id.0, param_types);
233                self.function_names.insert(id.0, func.name.clone());
234            }
235        }
236    }

238    fn declare_variable(
239        &mut self,
240        name: &str,
241        data_type: DataType,
242        is_mutable: bool,
243        span: Option<TextSpan>,
244    ) -> ResolvedBinding {
245        let frame_idx = self.frames.len() - 1;
246        let is_global = self.frames[frame_idx].is_global;
247
248        if let Some(existing) = self.frames[frame_idx].scopes.last().unwrap().variables.get(name) {
249            let slot = existing.slot;
250            self.error_at(format!("Variable '{}' already declared in this scope", name), span);
251            return if is_global { ResolvedBinding::Global(slot) } else { ResolvedBinding::Local(slot) };
252        }
253
254        let slot = SlotIndex(self.frames[frame_idx].next_slot);
255        self.frames[frame_idx].next_slot += 1;
256        self.frames[frame_idx]
257            .scopes
258            .last_mut()
259            .unwrap()
260            .variables
261            .insert(name.to_string(), VarInfo { slot, data_type, is_mutable });
262
263        if is_global { ResolvedBinding::Global(slot) } else { ResolvedBinding::Local(slot) }
264    }
265
266    /// Searches the current frame's scope stack, then falls back to the
267    /// global frame only — nested functions never see an intermediate
268    /// enclosing function's locals (see design doc §9).
269    fn lookup_variable(&self, name: &str) -> Option<(ResolvedBinding, DataType, bool)> {
270        let frame = self.frames.last().unwrap();
271        for scope in frame.scopes.iter().rev() {
272            if let Some(info) = scope.variables.get(name) {
273                let binding = if frame.is_global {
274                    ResolvedBinding::Global(info.slot)
275                } else {
276                    ResolvedBinding::Local(info.slot)
277                };
278                return Some((binding, info.data_type, info.is_mutable));
279            }
280        }
281        if !frame.is_global {
282            for scope in self.frames[0].scopes.iter().rev() {
283                if let Some(info) = scope.variables.get(name) {
284                    return Some((ResolvedBinding::Global(info.slot), info.data_type, info.is_mutable));
285                }
286            }
287        }
288        None
289    }
290
291    fn lookup_function(&self, name: &str) -> Option<FunctionId> {
292        let frame = self.frames.last().unwrap();
293        for scope in frame.scopes.iter().rev() {
294            if let Some(&id) = scope.functions.get(name) {
295                return Some(id);
296            }
297        }
298        if !frame.is_global {
299            for scope in self.frames[0].scopes.iter().rev() {
300                if let Some(&id) = scope.functions.get(name) {
301                    return Some(id);
302                }
303            }
304        }
305        None
306    }
307
308    fn visible_variable_names(&self) -> Vec<String> {
309        let mut names = Vec::new();
310        let mut seen = std::collections::HashSet::new();
311        let frame = self.frames.last().unwrap();
312        for scope in frame.scopes.iter().rev() {
313            for name in scope.variables.keys() {
314                if seen.insert(name.clone()) {
315                    names.push(name.clone());
316                }
317            }
318        }
319        if !frame.is_global {
320            for scope in self.frames[0].scopes.iter().rev() {
321                for name in scope.variables.keys() {
322                    if seen.insert(name.clone()) {
323                        names.push(name.clone());
324                    }
325                }
326            }
327        }
328        names
329    }

333    /// Resolves `expr`, records its type on the node, and returns it.
334    fn visit_expression_typed(&mut self, expr: &ASTExpression) -> DataType {
335        let ty = self.resolve_expr_type(expr);
336        expr.resolved_type.set(Some(ty));
337        ty
338    }
339
340    fn resolve_expr_type(&mut self, expression: &ASTExpression) -> DataType {
341        match &expression.kind {
342            ASTExpressionKind::Number(n) => n.value.get_type(),
343            ASTExpressionKind::Binary(bin) => self.resolve_binary(bin),
344            ASTExpressionKind::Paranthesized(p) => self.visit_expression_typed(&p.expression),
345            ASTExpressionKind::Unary(u) => self.resolve_unary(u),
346            ASTExpressionKind::PostfixUnary(p) => self.resolve_postfix(p),
347            ASTExpressionKind::Identifier(ident) => self.resolve_identifier(ident),
348            ASTExpressionKind::FunctionCall(call) => self.resolve_call(call),
349        }
350    }
351
352    fn resolve_identifier(&mut self, ident: &ASTIdentifierExpression) -> DataType {
353        let span = ident.token.as_ref().map(|t| t.span.clone());
354        match self.lookup_variable(&ident.name) {
355            Some((binding, data_type, _)) => {
356                ident.binding.set(Some(binding));
357                data_type
358            }
359            None => {
360                let message = format!("Variable '{}' not found", ident.name);
361                match closest_name(&ident.name, &self.visible_variable_names()) {
362                    Some(s) => self.error_with_suggestion(message, span, format!("did you mean '{}' ?", s)),
363                    None => self.error_at(message, span),
364                }
365                DataType::Any
366            }
367        }
368    }

370    /// Mirrors `Value::coerce_to_common_type`'s numeric-only cases (no
371    /// string concatenation) — shared by every arithmetic operator except `+`.
372    fn numeric_result(left: DataType, right: DataType) -> Option<DataType> {
373        use DataType::*;
374        match (left, right) {
375            (Integer, Integer) => Some(Integer),
376            (Float, Float) | (Integer, Float) | (Float, Integer) => Some(Float),
377            _ => None,
378        }
379    }
380
381    fn check_bitwise(&mut self, left: DataType, right: DataType, span: Option<TextSpan>, message: &str) -> DataType {
382        if left == DataType::Any || right == DataType::Any {
383            return DataType::Any;
384        }
385        // Value::to_integer() at runtime only fails for String.
386        if left == DataType::String || right == DataType::String {
387            self.error_at(message.to_string(), span);
388            return DataType::Any;
389        }
390        DataType::Integer
391    }

393    fn is_comparable_pair(left: DataType, right: DataType) -> bool {
394        left == right || matches!((left, right), (DataType::Integer, DataType::Float) | (DataType::Float, DataType::Integer))
395    }
396
397    fn check_equality(&mut self, left: DataType, right: DataType, span: Option<TextSpan>) -> DataType {
398        if left == DataType::Any || right == DataType::Any {
399            return DataType::Boolean;
400        }
401        if Self::is_comparable_pair(left, right) {
402            DataType::Boolean
403        } else {
404            self.error_at(format!("Cannot compare {:?} and {:?} for equality", left, right), span);
405            DataType::Any
406        }
407    }
408
409    fn check_ordering(&mut self, left: DataType, right: DataType, span: Option<TextSpan>) -> DataType {
410        if left == DataType::Any || right == DataType::Any {
411            return DataType::Boolean;
412        }
413        if Self::is_comparable_pair(left, right) {
414            DataType::Boolean
415        } else {
416            self.error_at(format!("Cannot compare {:?} and {:?}", left, right), span);
417            DataType::Any
418        }
419    }

421    fn resolve_binary(&mut self, bin: &ASTBinaryExpression) -> DataType {
422        let left = self.visit_expression_typed(&bin.left);
423        let right = self.visit_expression_typed(&bin.right);
424        let span = Some(bin.operator.token.span.clone());
425        use ASTBinaryOperatorKind::*;
426        use DataType::Any;
427
428        match bin.operator.kind {
429            // to_boolean() never fails, regardless of operand type.
430            LogicalAnd | LogicalOr => DataType::Boolean,
431
432            Plus => {
433                if left == Any || right == Any {
434                    Any
435                } else if left == DataType::String && right == DataType::String {
436                    DataType::String
437                } else if let Some(t) = Self::numeric_result(left, right) {
438                    t
439                } else {
440                    self.error_at(format!("Cannot add {:?} and {:?}", left, right), span);
441                    Any
442                }
443            }
444            Minus => {
445                if left == Any || right == Any {
446                    Any
447                } else if let Some(t) = Self::numeric_result(left, right) {
448                    t
449                } else {
450                    self.error_at(format!("Cannot subtract {:?} from {:?}", right, left), span);
451                    Any
452                }
453            }
454            Multiply => {
455                if left == Any || right == Any {
456                    Any
457                } else if let Some(t) = Self::numeric_result(left, right) {
458                    t
459                } else {
460                    self.error_at(format!("Cannot multiply {:?} and {:?}", left, right), span);
461                    Any
462                }
463            }
464            Divide => {
465                if left == Any || right == Any {
466                    Any
467                } else if let Some(t) = Self::numeric_result(left, right) {
468                    t
469                } else {
470                    self.error_at(format!("Cannot divide {:?} by {:?}", left, right), span);
471                    Any
472                }
473            }
474            Modulo => {
475                if left == Any || right == Any {
476                    Any
477                } else if let Some(t) = Self::numeric_result(left, right) {
478                    t
479                } else {
480                    self.error_at(format!("Cannot compute modulo of {:?} and {:?}", left, right), span);
481                    Any
482                }
483            }
484            Exponentiation => {
485                if left == Any || right == Any {
486                    Any
487                } else if let Some(t) = Self::numeric_result(left, right) {
488                    t
489                } else {
490                    self.error_at(format!("Cannot exponentiate {:?} and {:?}", left, right), span);
491                    Any
492                }
493            }
494            BitwiseAnd => self.check_bitwise(left, right, span, "Bitwise AND requires integer operands"),
495            BitwiseOr => self.check_bitwise(left, right, span, "Bitwise OR requires integer operands"),
496            BitwiseXor => self.check_bitwise(left, right, span, "Bitwise XOR requires integer operands"),
497            LeftShift => self.check_bitwise(left, right, span, "Left shift requires integer operands"),
498            RightShift => self.check_bitwise(left, right, span, "Right shift requires integer operands"),
499            Equal | NotEqual => self.check_equality(left, right, span),
500            Less | Greater | LessEqual | GreaterEqual => self.check_ordering(left, right, span),
501        }
502    }

504    fn resolve_unary(&mut self, u: &ASTUnaryExpression) -> DataType {
505        match u.operator.kind {
506            ASTUnaryOperatorKind::Increment | ASTUnaryOperatorKind::Decrement => {
507                self.resolve_inc_dec(&u.operand, matches!(u.operator.kind, ASTUnaryOperatorKind::Increment), false)
508            }
509            ASTUnaryOperatorKind::Plus => {
510                // Runtime passthrough: valid (and returns the operand type)
511                // for every type, never errors.
512                self.visit_expression_typed(&u.operand)
513            }
514            ASTUnaryOperatorKind::Minus => {
515                let ty = self.visit_expression_typed(&u.operand);
516                match ty {
517                    DataType::Any | DataType::Integer | DataType::Float => ty,
518                    other => {
519                        self.error(format!("Cannot negate {:?}", other));
520                        DataType::Any
521                    }
522                }
523            }
524            ASTUnaryOperatorKind::LogicalNot => {
525                self.visit_expression_typed(&u.operand);
526                DataType::Boolean
527            }
528        }
529    }

531    fn resolve_postfix(&mut self, p: &ASTPostfixUnaryExpression) -> DataType {
532        self.resolve_inc_dec(&p.operand, matches!(p.operator.kind, ASTUnaryOperatorKind::Increment), true)
533    }

535    fn resolve_inc_dec(&mut self, operand: &ASTExpression, is_increment: bool, is_postfix: bool) -> DataType {
536        let ident = match &operand.kind {
537            ASTExpressionKind::Identifier(ident) => ident,
538            _ => {
539                let prefix = if is_postfix { "Postfix " } else { "" };
540                self.error(format!("{}Increment/Decrement can only be applied to variables", prefix));
541                return DataType::Any;
542            }
543        };
544
545        if let Some((_, _, is_mutable)) = self.lookup_variable(&ident.name) {
546            if !is_mutable {
547                self.error(format!("Cannot assign to immutable variable '{}'", ident.name));
548            }
549        }
550
551        let ty = self.visit_expression_typed(operand);
552        match ty {
553            DataType::Any | DataType::Integer | DataType::Float => ty,
554            other => {
555                let verb = if is_increment { "increment" } else { "decrement" };
556                self.error(format!("Cannot {} {:?}", verb, other));
557                DataType::Any
558            }
559        }
560    }

562    fn resolve_call(&mut self, call: &ASTFunctionCallExpression) -> DataType {
563        let span = call.token.as_ref().map(|t| t.span.clone());
564        let arg_types: Vec<DataType> = call.arguments.iter().map(|a| self.visit_expression_typed(a)).collect();
565
566        if let Some(builtin) = builtin_named(&call.name) {
567            call.resolved_call.set(Some(ResolvedCall::Builtin(builtin)));
568            return self.resolve_builtin_call(builtin, &arg_types, span);
569        }
570
571        match self.lookup_function(&call.name) {
572            Some(id) => {
573                call.resolved_call.set(Some(ResolvedCall::User(id)));
574                let expected = self.function_param_types.get(&id.0).cloned().unwrap_or_default();
575                let name = self.function_names.get(&id.0).cloned().unwrap_or_else(|| call.name.clone());
576                if arg_types.len() != expected.len() {
577                    self.error_at(
578                        format!("Function '{}' expected {} arguments but got {}", name, expected.len(), arg_types.len()),
579                        span,
580                    );
581                    return DataType::Any;
582                }
583                for (i, (&expected_ty, &actual_ty)) in expected.iter().zip(arg_types.iter()).enumerate() {
584                    let widening_ok = expected_ty == DataType::Float && actual_ty == DataType::Integer;
585                    if expected_ty != DataType::Any && actual_ty != DataType::Any && expected_ty != actual_ty && !widening_ok {
586                        self.error_at(
587                            format!(
588                                "Type mismatch: argument {} to '{}' expected {:?} but got {:?}",
589                                i + 1, name, expected_ty, actual_ty
590                            ),
591                            span.clone(),
592                        );
593                    }
594                }
595                if self.currently_resolving.contains(&id.0) {
596                    // Direct or mutual recursion: this function's return
597                    // type isn't known yet.
598                    return DataType::Any;
599                }
600                self.function_return_types.get(&id.0).copied().unwrap_or(DataType::Any)
601            }
602            None => {
603                let mut names: Vec<String> = vec!["print".to_string(), "max".to_string(), "min".to_string()];
604                names.extend(self.function_names.values().cloned());
605                let message = format!("Unknown function: '{}'", call.name);
606                match closest_name(&call.name, &names) {
607                    Some(s) => self.error_with_suggestion(message, span, format!("did you mean '{}' ?", s)),
608                    None => self.error_at(message, span),
609                }
610                DataType::Any
611            }
612        }
613    }

615    fn resolve_builtin_call(&mut self, builtin: BuiltinFn, arg_types: &[DataType], span: Option<TextSpan>) -> DataType {
616        match builtin {
617            // print() never produces a usable value.
618            BuiltinFn::Print => DataType::Any,
619            BuiltinFn::Max | BuiltinFn::Min => {
620                if arg_types.is_empty() {
621                    let verb = if matches!(builtin, BuiltinFn::Max) { "max" } else { "min" };
622                    self.error_at(format!("{}() requires at least one argument", verb), span);
623                    return DataType::Any;
624                }
625                // ponytail: mismatched-argument-type errors inside max/min
626                // stay a runtime concern (Value::compare's error text);
627                // statically we just fall back to Any for a mixed call.
628                Self::combine_types(arg_types)
629            }
630        }
631    }
632
633    /// Unifies a set of types the same way assignment widening does
634    /// (Integer+Float -> Float); anything else collapses to `Any`.
635    fn combine_types(types: &[DataType]) -> DataType {
636        let mut result = match types.first() {
637            Some(t) => *t,
638            None => return DataType::Any,
639        };
640        for &t in &types[1..] {
641            result = match (result, t) {
642                (a, b) if a == b => a,
643                (DataType::Integer, DataType::Float) | (DataType::Float, DataType::Integer) => DataType::Float,
644                _ => return DataType::Any,
645            };
646        }
647        result
648    }

651
652    fn resolve_function_body(&mut self, func: &ASTFunctionDeclaration, id: FunctionId, emit_warnings: bool) -> DataType {
653        self.frames.push(Frame { is_global: false, next_slot: 0, scopes: vec![BlockScope::new()] });
654
655        let mut param_types = Vec::new();
656        let mut any_untyped = false;
657        for param in &func.parameters {
658            let ty = match param.type_annotation {
659                Some(t) => t,
660                None => {
661                    any_untyped = true;
662                    DataType::Any
663                }
664            };
665            let binding = self.declare_variable(&param.name, ty, true, None);
666            param.slot.set(Some(binding.slot()));
667            param_types.push(ty);
668        }
669        self.function_param_types.insert(id.0, param_types);
670        self.function_names.insert(id.0, func.name.clone());
671
672        if any_untyped && emit_warnings {
673            self.warn(format!(
674                "cannot statically infer a type for one or more parameters of '{}'; falling back to dynamic type checking",
675                func.name
676            ));
677        }
678
679        // Params and the body's own top-level declarations share one scope
680        // (matching the old evaluator: shadowing a parameter at the body's
681        // top level was already a "redeclared in this scope" error).
682        self.return_types_stack.push(Vec::new());
683        self.hoist_and_resolve(&func.body);
684        let seen_returns = self.return_types_stack.pop().unwrap();
685
686        let return_type = if !seen_returns.is_empty() {
687            Self::combine_types(&seen_returns)
688        } else {
689            Self::last_expression_type(&func.body).unwrap_or(DataType::Any)
690        };
691
692        let frame_size = self.frames.last().unwrap().next_slot;
693        func.frame_size.set(Some(frame_size));
694        self.frames.pop();
695
696        return_type
697    }
698
699    fn last_expression_type(body: &[ASTStatement]) -> Option<DataType> {
700        match &body.last()?.kind {
701            ASTStatementKind::Expression(e) => e.resolved_type.get(),
702            _ => None,
703        }
704    }
705
706    fn calls_function_by_name(statements: &[ASTStatement], name: &str) -> bool {
707        statements.iter().any(|s| Self::stmt_calls(s, name))
708    }
709
710    fn stmt_calls(stmt: &ASTStatement, name: &str) -> bool {
711        match &stmt.kind {
712            ASTStatementKind::Expression(e) => Self::expr_calls(e, name),
713            ASTStatementKind::VariableDeclaration(d) => Self::expr_calls(&d.initializer, name),
714            ASTStatementKind::Assignment(a) => Self::expr_calls(&a.value, name),
715            ASTStatementKind::Block(stmts) => Self::calls_function_by_name(stmts, name),
716            ASTStatementKind::IfStatement(i) => {
717                Self::expr_calls(&i.condition, name)
718                    || Self::calls_function_by_name(&i.then_branch, name)
719                    || i.else_branch.as_ref().is_some_and(|b| Self::calls_function_by_name(b, name))
720            }
721            // A nested fn's body is a separate scope; it doesn't count as
722            // the enclosing function calling itself.
723            ASTStatementKind::FunctionDeclaration(_) => false,
724            ASTStatementKind::ReturnStatement(r) => Self::expr_calls(&r.value, name),
725        }
726    }
727
728    fn expr_calls(expr: &ASTExpression, name: &str) -> bool {
729        match &expr.kind {
730            ASTExpressionKind::Number(_) | ASTExpressionKind::Identifier(_) => false,
731            ASTExpressionKind::Binary(b) => Self::expr_calls(&b.left, name) || Self::expr_calls(&b.right, name),
732            ASTExpressionKind::Paranthesized(p) => Self::expr_calls(&p.expression, name),
733            ASTExpressionKind::Unary(u) => Self::expr_calls(&u.operand, name),
734            ASTExpressionKind::PostfixUnary(p) => Self::expr_calls(&p.operand, name),
735            ASTExpressionKind::FunctionCall(c) => c.name == name || c.arguments.iter().any(|a| Self::expr_calls(a, name)),
736        }
737    }
738}

740impl ASTVisitor for Resolver {
741    // Bypassed entirely by our own `visit_expression` override below; kept
742    // only to satisfy the trait (it has no default implementation).
743    fn visit_number(&mut self, _number: &crate::ast::ASTNumberExpression) {}
744
745    fn visit_expression(&mut self, expression: &ASTExpression) {
746        self.visit_expression_typed(expression);
747    }
748
749    fn visit_variable_declaration(&mut self, decl: &ASTVariableDeclaration) {
750        let init_ty = self.visit_expression_typed(&decl.initializer);
751        let span = decl.name_span.as_ref().map(|t| t.span.clone());
752        let binding = self.declare_variable(&decl.name, init_ty, decl.is_mutable, span);
753        decl.binding.set(Some(binding));
754    }
755
756    fn visit_assignment(&mut self, assign: &ASTAssignment) {
757        let value_ty = self.visit_expression_typed(&assign.value);
758        let span = assign.name_span.as_ref().map(|t| t.span.clone());
759
760        match self.lookup_variable(&assign.name) {
761            None => {
762                let message = format!("Variable '{}' not found", assign.name);
763                match closest_name(&assign.name, &self.visible_variable_names()) {
764                    Some(s) => self.error_with_suggestion(message, span, format!("did you mean '{}' ?", s)),
765                    None => self.error_at(message, span),
766                }
767            }
768            Some((binding, declared_ty, is_mutable)) => {
769                assign.binding.set(Some(binding));
770                if !is_mutable {
771                    self.error_at(format!("Cannot assign to immutable variable '{}'", assign.name), span);
772                } else if value_ty != DataType::Any && declared_ty != DataType::Any && value_ty != declared_ty {
773                    let widening_ok = declared_ty == DataType::Float && value_ty == DataType::Integer;
774                    if widening_ok {
775                        assign.needs_float_widen.set(true);
776                    } else {
777                        self.error_at(
778                            format!(
779                                "Type mismatch: variable '{}' has type {:?}, cannot assign {:?}",
780                                assign.name, declared_ty, value_ty
781                            ),
782                            span,
783                        );
784                    }
785                }
786            }
787        }
788    }
789
790    fn visit_function_declaration(&mut self, func: &ASTFunctionDeclaration) {
791        let id = func.function_id.get().expect("function must be hoisted before its body is resolved");
792
793        self.currently_resolving.push(id.0);
794        let tentative = self.resolve_function_body(func, id, true);
795        self.currently_resolving.pop();
796        self.function_return_types.insert(id.0, tentative);
797
798        let final_return = if tentative != DataType::Any && Self::calls_function_by_name(&func.body, &func.name) {
799            let refined = self.resolve_function_body(func, id, false);
800            self.function_return_types.insert(id.0, refined);
801            refined
802        } else {
803            tentative
804        };
805
806        if final_return == DataType::Any {
807            self.warn(format!(
808                "cannot statically infer a return type for '{}'; falling back to dynamic type checking",
809                func.name
810            ));
811        }
812
813        func.inferred_return_type.set(Some(final_return));
814    }
815
816    fn visit_return_statement(&mut self, ret: &ASTReturnStatement) {
817        let ty = self.visit_expression_typed(&ret.value);
818        match self.return_types_stack.last_mut() {
819            Some(returns) => returns.push(ty),
820            None => self.error("'return' can only be used inside a function".to_string()),
821        }
822    }
823
824    fn visit_if_statement(&mut self, if_stmt: &ASTIfStatement) {
825        self.visit_expression_typed(&if_stmt.condition);
826        self.resolve_statements(&if_stmt.then_branch);
827        if let Some(else_branch) = &if_stmt.else_branch {
828            self.resolve_statements(else_branch);
829        }
830    }
831
832    fn visit_block(&mut self, statements: &Vec<ASTStatement>) {
833        self.resolve_statements(statements);
834    }
835}

843    fn resolve(source: &str) -> (Ast, Resolver) {
844        let mut lexer = Lexer::new(source);
845        let mut tokens = Vec::new();
846        while let Some(token) = lexer.next_token() {
847            tokens.push(token);
848        }
849        let mut parser = Parser::new(tokens);
850        let mut ast = Ast::new();
851        while let Some(stmt) = parser.next_statement() {
852            ast.add_statement(stmt);
853        }
854        assert!(parser.diagnostics.is_empty());
855
856        let mut resolver = Resolver::new();
857        resolver.resolve(&ast);
858        (ast, resolver)
859    }

865        assert_eq!(resolver.global_slot_count(), 3);

867        let mut slots = Vec::new();

870                slots.push(decl.binding.get().unwrap().slot().0);
