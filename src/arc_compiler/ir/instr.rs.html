1//! # IR Types
2//!
3//! Three-address code over virtual registers, organized into basic blocks
4//! with explicit terminators — the representation the Phase 4 optimization
5//! passes operate on. Not SSA: registers are *almost* single-assignment
6//! (lowering writes each register once per block; only `&&`/`||` join
7//! results are written in two different blocks), and locals/globals stay
8//! slot-addressed memory operations rather than being register-promoted.
9//! See `local-notes/phase4-ir-design.md`.
10
11use crate::ast::resolver::{BuiltinFn, SlotIndex};
12use crate::ast::types::Value;
13use crate::ast::ASTBinaryOperatorKind;
14
15/// A virtual register. Infinite supply, function-local.
16#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
17pub struct Reg(pub u32);
18
19/// A basic-block id — an index into [`IrFunction::blocks`].
20#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
21pub struct BlockId(pub u32);
22
23/// One IR operation plus the source byte-offset it originated from (same
24/// convention as [`Chunk::offsets`](crate::bytecode::chunk::Chunk::offsets),
25/// so bytecode lowering can rebuild the offset table and runtime
26/// diagnostics survive optimization).
27#[derive(Debug, Clone)]
28pub struct Instruction {
29    pub kind: InstrKind,
30    pub offset: u32,
31}
32
33#[derive(Debug, Clone)]
34pub enum InstrKind {
35    Const { dst: Reg, value: Value },
36    /// A non-logical binary operation (`&&`/`||` never reach the IR — they
37    /// lower to control flow). `can_error` is the shared purity fact every
38    /// pass keys off: `true` if this op could raise a runtime error
39    /// (division/modulo by zero, `**` overflow, or any operand typed `Any`),
40    /// in which case folding skips error results and DCE/CSE leave it alone.
41    Binary { dst: Reg, op: ASTBinaryOperatorKind, lhs: Reg, rhs: Reg, can_error: bool },
42    /// Arithmetic negation. Errors at runtime on non-numeric values, so it
43    /// carries its own `can_error` (true iff the operand's static type is
44    /// `Any`).
45    Negate { dst: Reg, src: Reg, can_error: bool },
46    /// Logical NOT with boolean coercion (`!x`). Never errors.
47    Not { dst: Reg, src: Reg },
48    /// Boolean coercion (`Value::to_boolean`), used by `&&`/`||` lowering.
49    /// Never errors.
50    ToBool { dst: Reg, src: Reg },
51    /// Resolver-decided int→float widening. Never errors.
52    ToFloat { dst: Reg, src: Reg },
53    LoadLocal { dst: Reg, slot: SlotIndex },
54    StoreLocal { slot: SlotIndex, src: Reg },
55    LoadGlobal { dst: Reg, slot: SlotIndex },
56    StoreGlobal { slot: SlotIndex, src: Reg },
57    Call { dst: Reg, function: u32, args: Vec<Reg> },
58    CallBuiltin { dst: Reg, builtin: BuiltinFn, args: Vec<Reg> },
59}
60
61#[derive(Debug, Clone)]
62pub enum Terminator {
63    Jump(BlockId),
64    Branch { cond: Reg, then_block: BlockId, else_block: BlockId },
65    Return(Reg),
66}
67
68#[derive(Debug)]
69pub struct Block {
70    pub instrs: Vec<Instruction>,
71    pub terminator: Terminator,
72    pub term_offset: u32,
73    /// Set by unreachable-block elimination. Dead blocks stay in the vec so
74    /// `BlockId`s remain stable; dump and bytecode lowering skip them.
75    pub dead: bool,
76}
77
78pub struct IrFunction {
79    pub name: String,
80    /// Local slots (parameters + declared variables) — the Resolver's
81    /// per-function frame size. Registers live *above* these when the
82    /// bytecode lowering spills them to slots.
83    pub frame_size: u32,
84    pub reg_count: u32,
85    /// Entry block is always `blocks[0]`.
86    pub blocks: Vec<Block>,
87}
88
89pub struct IrProgram {
90    pub script: IrFunction,
91    /// Indexed by `FunctionId.0`.
92    pub functions: Vec<IrFunction>,
93}
94
95impl InstrKind {
96    /// The register this instruction writes, if any.
97    pub fn dst(&self) -> Option<Reg> {
98        use InstrKind::*;
99        match self {
100            Const { dst, .. }
101            | Binary { dst, .. }
102            | Negate { dst, .. }
103            | Not { dst, .. }
104            | ToBool { dst, .. }
105            | ToFloat { dst, .. }
106            | LoadLocal { dst, .. }
107            | LoadGlobal { dst, .. }
108            | Call { dst, .. }
109            | CallBuiltin { dst, .. } => Some(*dst),
110            StoreLocal { .. } | StoreGlobal { .. } => None,
111        }
112    }
113
114    /// Visits every register this instruction *reads* (mutably, so passes
115    /// can rewrite operands in place).
116    pub fn for_each_use(&mut self, f: &mut dyn FnMut(&mut Reg)) {
117        use InstrKind::*;
118        match self {
119            Const { .. } | LoadLocal { .. } | LoadGlobal { .. } => {}
120            Binary { lhs, rhs, .. } => {
121                f(lhs);
122                f(rhs);
123            }
124            Negate { src, .. } | Not { src, .. } | ToBool { src, .. } | ToFloat { src, .. } => f(src),
125            StoreLocal { src, .. } | StoreGlobal { src, .. } => f(src),
126            Call { args, .. } | CallBuiltin { args, .. } => {
127                for arg in args {
128                    f(arg);
129                }
130            }
131        }
132    }
133
134    /// Whether the instruction is removable when its result is unused:
135    /// no side effects *and* provably unable to raise a runtime error.
136    /// Deleting a potential runtime error is a behavior change, so
137    /// `Binary`/`Negate` consult the `can_error` fact computed at lowering.
138    pub fn is_pure(&self) -> bool {
139        use InstrKind::*;
140        match self {
141            Const { .. } | Not { .. } | ToBool { .. } | ToFloat { .. } | LoadLocal { .. } | LoadGlobal { .. } => true,
142            Binary { can_error, .. } | Negate { can_error, .. } => !can_error,
143            StoreLocal { .. } | StoreGlobal { .. } | Call { .. } | CallBuiltin { .. } => false,
144        }
145    }
146}
147
148impl IrFunction {
149    /// Live (non-dead-block) instruction count — the metric the exit
150    /// criterion's "concrete reduction" is measured in. Terminators aren't
151    /// counted; passes don't remove control flow except with dead blocks.
152    pub fn instr_count(&self) -> usize {
153        self.blocks.iter().filter(|b| !b.dead).map(|b| b.instrs.len()).sum()
154    }
155}
156
157impl IrProgram {
158    pub fn instr_count(&self) -> usize {
159        self.script.instr_count() + self.functions.iter().map(IrFunction::instr_count).sum::<usize>()
160    }
