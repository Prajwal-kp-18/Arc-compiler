1//! # IR Dump
2//!
3//! Text pretty-printer for [`IrProgram`] — the disassembler's sibling,
4//! wired up behind `--dump-ir` (post-lowering) and `--dump-ir=opt`
5//! (post-passes). The before/after diff of these two is the Phase 4 exit
6//! criterion's demo artifact.
7
8use crate::ast::resolver::BuiltinFn;
9use crate::ast::ASTBinaryOperatorKind;

11use super::instr::{InstrKind, IrFunction, IrProgram, Terminator};

13pub fn dump_program(program: &IrProgram) -> String {
14    let mut out = dump_function(&program.script, "script");
15    for (id, function) in program.functions.iter().enumerate() {
16        out.push('\n');
17        out.push_str(&dump_function(function, &format!("fn#{} {}", id, function.name)));
18    }
19    out
20}
21
22pub fn dump_function(f: &IrFunction, name: &str) -> String {
23    let mut out = format!("== {} (frame_size={}, {} instrs) ==\n", name, f.frame_size, f.instr_count());
24    for (id, block) in f.blocks.iter().enumerate() {
25        if block.dead {
26            continue;
27        }
28        out.push_str(&format!("b{}:\n", id));
29        for instr in &block.instrs {
30            out.push_str(&format!("    {}\n", render(&instr.kind)));
31        }
32        let term = match &block.terminator {
33            Terminator::Jump(b) => format!("jump b{}", b.0),
34            Terminator::Branch { cond, then_block, else_block } => {
35                format!("branch r{} ? b{} : b{}", cond.0, then_block.0, else_block.0)
36            }
37            Terminator::Return(r) => format!("return r{}", r.0),
38        };
39        out.push_str(&format!("    {}\n", term));
40    }
41    out
42}

44fn render(kind: &InstrKind) -> String {
45    use InstrKind::*;
46    match kind {
47        Const { dst, value } => format!("r{} = const {:?}", dst.0, value),
48        Binary { dst, op, lhs, rhs, can_error } => format!(
49            "r{} = {} r{}, r{}{}",
50            dst.0,
51            mnemonic(op),
52            lhs.0,
53            rhs.0,
54            if *can_error { " !" } else { "" }
55        ),
56        Negate { dst, src, can_error } => {
57            format!("r{} = neg r{}{}", dst.0, src.0, if *can_error { " !" } else { "" })
58        }
59        Not { dst, src } => format!("r{} = not r{}", dst.0, src.0),
60        ToBool { dst, src } => format!("r{} = to_bool r{}", dst.0, src.0),
61        ToFloat { dst, src } => format!("r{} = to_float r{}", dst.0, src.0),
62        LoadLocal { dst, slot } => format!("r{} = load_local {}", dst.0, slot.0),
63        StoreLocal { slot, src } => format!("store_local {} = r{}", slot.0, src.0),
64        LoadGlobal { dst, slot } => format!("r{} = load_global {}", dst.0, slot.0),
65        StoreGlobal { slot, src } => format!("store_global {} = r{}", slot.0, src.0),
66        Call { dst, function, args } => format!("r{} = call fn#{}({})", dst.0, function, regs(args)),
67        CallBuiltin { dst, builtin, args } => {
68            let name = match builtin {
69                BuiltinFn::Print => "print",
70                BuiltinFn::Max => "max",
71                BuiltinFn::Min => "min",
72            };
73            format!("r{} = call {}({})", dst.0, name, regs(args))
74        }
75    }
76}
77
78fn regs(args: &[super::instr::Reg]) -> String {
79    args.iter().map(|r| format!("r{}", r.0)).collect::<Vec<_>>().join(", ")
80}

82fn mnemonic(op: &ASTBinaryOperatorKind) -> &'static str {
83    use ASTBinaryOperatorKind::*;
84    match op {
85        Plus => "add",
86        Minus => "sub",
87        Multiply => "mul",
88        Divide => "div",
89        Modulo => "mod",
90        Exponentiation => "pow",
91        BitwiseAnd => "band",
92        BitwiseOr => "bor",
93        BitwiseXor => "bxor",
94        LeftShift => "shl",
95        RightShift => "shr",
96        Equal => "eq",
97        NotEqual => "ne",
98        Less => "lt",
99        Greater => "gt",
100        LessEqual => "le",
101        GreaterEqual => "ge",
102        LogicalAnd | LogicalOr => unreachable!("logical operators lower to control flow"),
103    }
104}
