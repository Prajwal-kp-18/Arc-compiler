1//! # Slot Liveness Analysis
2//!
3//! Backward dataflow over local frame slots (`LoadLocal`/`StoreLocal`): a
4//! slot is live at a program point if its current value may be read before
5//! being overwritten along some path to function exit. Globals are out of
6//! scope — top-level `let` resolves to `ResolvedBinding::Global`, and
7//! `StoreGlobal` is conservatively always live (see [`super::passes::dse`]).
8//! Feeds dead-store elimination and `--dump-liveness`.
9//! See `local-notes/phase4.6-liveness-design.md`.
10
11use std::collections::HashSet;
12
13use super::instr::{Block, BlockId, InstrKind, IrFunction, Terminator};
14
15/// Live-in/live-out slot sets per block, indexed by `BlockId.0 as usize`.
16/// Dead blocks keep empty sets.
17pub struct Liveness {
18    pub live_in: Vec<HashSet<u32>>,
19    pub live_out: Vec<HashSet<u32>>,
20}
21
22/// Runs the analysis to a fixpoint over `f`'s CFG (loop back-edges included
23/// via ordinary successor edges, so loop-carried slots converge correctly).
24pub fn analyze(f: &IrFunction) -> Liveness {
25    let n = f.blocks.len();
26    let mut live_in: Vec<HashSet<u32>> = vec![HashSet::new(); n];
27    let mut live_out: Vec<HashSet<u32>> = vec![HashSet::new(); n];
28
29    loop {
30        let mut changed = false;
31        for id in (0..n).rev() {
32            if f.blocks[id].dead {
33                continue;
34            }
35            let mut out = HashSet::new();
36            for succ in successors(&f.blocks[id]) {
37                out.extend(live_in[succ.0 as usize].iter().copied());
38            }
39            let inp = transfer(&f.blocks[id], &out);
40            if out != live_out[id] {
41                live_out[id] = out;
42                changed = true;
43            }
44            if inp != live_in[id] {
45                live_in[id] = inp;
46                changed = true;
47            }
48        }
49        if !changed {
50            break;
51        }
52    }
53
54    Liveness { live_in, live_out }
55}
56
57/// Per-instruction live-after sets within `block`, index-aligned with
58/// `block.instrs`, walked backward from `live_out`. Liveness is a
59/// block-boundary fact; DSE and the dump both need it *within* a block too.
60pub fn live_after_each(block: &Block, live_out: &HashSet<u32>) -> Vec<HashSet<u32>> {
61    let mut result = vec![HashSet::new(); block.instrs.len()];
62    let mut live = live_out.clone();
63    for (idx, instr) in block.instrs.iter().enumerate().rev() {
64        result[idx] = live.clone();
65        apply_transfer(&instr.kind, &mut live);
66    }
67    result
68}
69
70fn successors(block: &Block) -> Vec<BlockId> {
71    match &block.terminator {
72        Terminator::Jump(b) => vec![*b],
73        Terminator::Branch { then_block, else_block, .. } => vec![*then_block, *else_block],
74        Terminator::Return(_) => vec![],
75    }
76}
77
78fn transfer(block: &Block, live_out: &HashSet<u32>) -> HashSet<u32> {
79    let mut live = live_out.clone();
80    for instr in block.instrs.iter().rev() {
81        apply_transfer(&instr.kind, &mut live);
82    }
83    live
84}
85
86/// `StoreLocal` kills its slot (def); `LoadLocal` generates its slot (use).
87/// Every other instruction kind — register ops, globals, calls — is
88/// transparent to slot liveness.
89fn apply_transfer(kind: &InstrKind, live: &mut HashSet<u32>) {
90    match kind {
91        InstrKind::StoreLocal { slot, .. } => {
92            live.remove(&slot.0);
93        }
94        InstrKind::LoadLocal { slot, .. } => {
95            live.insert(slot.0);
96        }
97        _ => {}
98    }
99}
