1//! # IR
2//!
3//! Phase 4 of the compiler pipeline: a three-address-code intermediate
4//! representation with basic blocks and explicit control flow, sitting
5//! between the resolved AST and codegen. [`lower`] builds it, [`passes`]
6//! optimizes it (constant folding, DCE, local CSE — all gated on preserving
7//! behavior *including runtime errors*), [`to_bytecode`] lowers it back to
8//! chunks so the existing VM executes it (`--opt`), and [`dump`] prints it
9//! (`--dump-ir`, `--dump-ir=opt`). See `local-notes/phase4-ir-design.md`.
10

30    fn lower_source(source: &str) -> IrProgram {
31        let mut lexer = Lexer::new(source);
32        let mut tokens = Vec::new();
33        while let Some(token) = lexer.next_token() {
34            tokens.push(token);
35        }
36        let mut parser = Parser::new(tokens);
37        let mut ast = Ast::new();
38        while let Some(stmt) = parser.next_statement() {
39            ast.add_statement(stmt);
40        }
41        assert!(parser.diagnostics.is_empty(), "parse errors: {:?}", parser.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>());
42
43        let mut resolver = Resolver::new();
44        resolver.resolve(&ast);
45        assert!(!resolver.has_errors(), "resolve errors: {:?}", resolver.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>());
46
47        lower::IrLowering::new(resolver.function_count()).lower(&ast)
48    }

50    /// Counts live instructions matching `pred` across the whole program.
51    fn count_instrs(program: &IrProgram, pred: impl Fn(&InstrKind) -> bool) -> usize {
52        std::iter::once(&program.script)
53            .chain(program.functions.iter())
54            .flat_map(|f| f.blocks.iter().filter(|b| !b.dead))
55            .flat_map(|b| b.instrs.iter())
56            .filter(|i| pred(&i.kind))
57            .count()
58    }

60    /// Runs a source program through tree-walker and the *optimized* IR
61    /// pipeline, asserting identical last_value and error messages.
62    fn assert_optimized_parity(source: &str) {
63        let mut lexer = Lexer::new(source);
64        let mut tokens = Vec::new();
65        while let Some(token) = lexer.next_token() {
66            tokens.push(token);
67        }
68        let mut parser = Parser::new(tokens);
69        let mut ast = Ast::new();
70        while let Some(stmt) = parser.next_statement() {
71            ast.add_statement(stmt);
72        }
73        let mut resolver = Resolver::new();
74        resolver.resolve(&ast);
75        assert!(!resolver.has_errors());
76
77        let mut evaluator = ASTEvaluator::new(resolver.global_slot_count());
78        evaluator.execute(&ast);
79
80        let mut program = lower::IrLowering::new(resolver.function_count()).lower(&ast);
81        passes::optimize(&mut program);
82        let compiled = to_bytecode::program_to_bytecode(&program);
83        let mut vm = VM::new(&compiled, resolver.global_slot_count());
84        vm.run();
85
86        assert_eq!(evaluator.last_value, vm.last_value, "last_value diverged on: {}", source);
87        let tree_errors: Vec<_> = evaluator.errors.iter().map(|d| d.message.clone()).collect();
88        let vm_errors: Vec<_> = vm.errors.iter().map(|d| d.message.clone()).collect();
89        assert_eq!(tree_errors, vm_errors, "errors diverged on: {}", source);
90    }

94    #[test]
95    fn test_optimized_pipeline_parity_on_golden_programs() {
96        let programs = [
97            "1 + 2 * 3;",
98            "2 ** 10;",
99            "let x = 1; x = x + 1; x;",
100            "let x = 1.5; x = 2; x + 0.5;",
101            "true && false || !true;",
102            "let r = 0; if true { r = 1; } else { r = 2; } r;",
103            "let r = 0; if false { r = 1; } else { r = 2; } r;",
104            "let v = 5; { let v = 99; } v;",
105            "fn add(a: Int, b: Int) { return a + b; } add(2, 3);",
106            "fn fact(n) { if n <= 1 { return 1; } else { return n * fact(n - 1); } } fact(10);",
107            "fn is_even(n) { if n == 0 { return true; } else { return is_odd(n - 1); } } \
108             fn is_odd(n) { if n == 0 { return false; } else { return is_even(n - 1); } } \
109             is_even(10);",
110            "let counter = 10; ++counter; counter++; --counter; counter--; counter;",
111            "min(4, 2, 9, 1); max(4, 2, 9, 1);",
112            "fn f() { 7; } f();",
113            "fn f() { return 1; 2; } f();", // unreachable code after return
114            "1 / 0;",
115            "10 % 0;",
116            r#"print(print("x"));"#,
117        ];
118        for source in programs {
119            assert_optimized_parity(source);
120        }
121    }

125    #[test]
126    fn test_fold_collapses_constant_arithmetic() {
127        let mut program = lower_source("1 + 2 * 3;");
128        passes::optimize(&mut program);
129        assert_eq!(count_instrs(&program, |k| matches!(k, InstrKind::Binary { .. })), 0);
130        // The whole script is one Const 7 feeding the return.
131        assert_eq!(program.instr_count(), 1, "\n{}", dump::dump_program(&program));
132        assert!(count_instrs(&program, |k| matches!(k, InstrKind::Const { value: Value::Integer(7), .. })) == 1);
133    }

150    #[test]
151    fn test_fold_constant_branch_kills_untaken_arm() {
152        let mut program = lower_source(r#"if true { print("a"); } else { print("b"); }"#);
153        passes::optimize(&mut program);
154        // Only the taken arm's print survives.
155        assert_eq!(count_instrs(&program, |k| matches!(k, InstrKind::CallBuiltin { .. })), 1);
156        // And no branch remains anywhere.
157        let text = dump::dump_program(&program);
158        assert!(!text.contains("branch"), "\n{}", text);

163    #[test]
164    fn test_dce_removes_unused_pure_result() {
165        let mut program = lower_source("fn f(a: Int) { a + 1; return 2; } f(1);");
166        passes::optimize(&mut program);
167        // `a + 1` is pure (typed Int, non-partial op) and unused -> gone,
168        // along with its operand loads.
169        let f = &program.functions[0];
170        assert_eq!(
171            f.blocks.iter().filter(|b| !b.dead).map(|b| b.instrs.len()).sum::<usize>(),
172            1, // just `const 2`
173            "\n{}",
174            dump::dump_program(&program)
175        );
176    }

178    #[test]
179    fn test_dce_keeps_unused_but_erroring_op() {
180        let mut program = lower_source("fn f(n: Int) { 10 % n; return 2; } f(1);");
181        passes::optimize(&mut program);
182        // `10 % n` can raise "Modulo by zero" — deleting it would change
183        // behavior for f(0), so it stays.
184        let in_f = |k: &InstrKind| matches!(k, InstrKind::Binary { .. });
185        assert_eq!(count_instrs(&program, in_f), 1);
186    }

190    #[test]
191    fn test_cse_dedupes_repeated_expression() {
192        let mut program = lower_source("fn f(a: Int, b: Int) { return (a * b) + (a * b); } f(2, 3);");
193        passes::optimize(&mut program);
194        let f = &program.functions[0];
195        let muls = f
196            .blocks
197            .iter()
198            .filter(|b| !b.dead)
199            .flat_map(|b| b.instrs.iter())
200            .filter(|i| matches!(i.kind, InstrKind::Binary { op: crate::ast::ASTBinaryOperatorKind::Multiply, .. }))
201            .count();
202        assert_eq!(muls, 1, "\n{}", dump::dump_program(&program));
203    }

217    fn test_passes_reduce_instruction_count() {
218        let source = "let a = 2 * 3 + 4; print(1 + 2 * 3, 2 ** 10, (4 - 2) * (4 - 2));";

220        let mut after = lower_source(source);

223            after.instr_count() < before.instr_count(),
224            "expected a reduction, got {} -> {}\nbefore:\n{}\nafter:\n{}",
