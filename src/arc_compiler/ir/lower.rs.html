1//! # AST → IR Lowering
2//!
3//! Structurally the sibling of `bytecode/compiler.rs`: the same infallible
4//! walk over the Resolver-annotated AST, emitting three-address instructions
5//! into basic blocks instead of bytes into a chunk. Control flow (`if`,
6//! `&&`, `||`) becomes explicit `Branch`/`Jump` edges; statement-level
7//! results are simply unused registers (which is what gives DCE something
8//! to delete).
9//!
10//! Block creation order guarantees every CFG edge targets a *higher* block
11//! id — Arc has no loops, so CFGs are DAGs and the bytecode lowering can rely
12//! on forward-only jumps. (`// CFG is a DAG today` — re-audit when `while`
13//! lands.)
14

25/// One function's blocks while they're being built (terminators arrive when
26/// a block is sealed).
27struct FnBuilder {
28    name: String,
29    frame_size: u32,
30    blocks: Vec<PendingBlock>,
31    current: usize,
32    next_reg: u32,
33}
34
35struct PendingBlock {
36    instrs: Vec<Instruction>,
37    terminator: Option<(Terminator, u32)>,
38}
39
40pub struct IrLowering {
41    functions: Vec<Option<IrFunction>>,
42    stack: Vec<FnBuilder>,
43    current_offset: u32,
44}
45
46impl IrLowering {
47    pub fn new(function_count: u32) -> Self {
48        Self {
49            functions: (0..function_count).map(|_| None).collect(),
50            stack: Vec::new(),
51            current_offset: 0,
52        }
53    }
54
55    pub fn lower(mut self, ast: &Ast) -> IrProgram {
56        self.push_function("script".to_string(), 0);
57        self.lower_body(&ast.statements);
58        let script = self.pop_function();
59        IrProgram {
60            script,
61            functions: self
62                .functions
63                .into_iter()
64                .map(|f| f.expect("Resolver guarantees every FunctionId has a declaration"))
65                .collect(),
66        }
67    }

71    fn push_function(&mut self, name: String, frame_size: u32) {
72        self.stack.push(FnBuilder {
73            name,
74            frame_size,
75            blocks: vec![PendingBlock { instrs: Vec::new(), terminator: None }],
76            current: 0,
77            next_reg: 0,
78        });
79    }
80
81    fn pop_function(&mut self) -> IrFunction {
82        let mut builder = self.stack.pop().expect("function builder was pushed");
83        // Any still-unsealed block at body end (detached unreachable code,
84        // or a join both of whose arms returned) gets a synthetic
85        // `return Unit` so the type is total — unreachable-block
86        // elimination drops the dead ones.
87        let offset = self.current_offset;
88        for block in &mut builder.blocks {
89            if block.terminator.is_none() {
90                let unit = Reg(builder.next_reg);
91                builder.next_reg += 1;
92                block.instrs.push(Instruction { kind: InstrKind::Const { dst: unit, value: Value::Unit }, offset });
93                block.terminator = Some((Terminator::Return(unit), offset));
94            }
95        }
96        IrFunction {
97            name: builder.name,
98            frame_size: builder.frame_size,
99            reg_count: builder.next_reg,
100            blocks: builder
101                .blocks
102                .into_iter()
103                .map(|b| {
104                    let (terminator, term_offset) = b.terminator.expect("every block is sealed");
105                    Block { instrs: b.instrs, terminator, term_offset, dead: false }
106                })
107                .collect(),
108        }
109    }
110
111    fn builder(&mut self) -> &mut FnBuilder {
112        self.stack.last_mut().expect("always lowering inside some function")
113    }
114
115    fn fresh_reg(&mut self) -> Reg {
116        let b = self.builder();
117        let r = Reg(b.next_reg);
118        b.next_reg += 1;
119        r
120    }
121
122    /// Creates a new (empty, unsealed) block and returns its id. Does not
123    /// switch to it.
124    fn new_block(&mut self) -> BlockId {
125        let b = self.builder();
126        b.blocks.push(PendingBlock { instrs: Vec::new(), terminator: None });
127        BlockId((b.blocks.len() - 1) as u32)
128    }
129
130    fn switch_to(&mut self, block: BlockId) {
131        self.builder().current = block.0 as usize;
132    }
133
134    fn emit(&mut self, kind: InstrKind) {
135        let offset = self.current_offset;
136        let b = self.builder();
137        let current = b.current;
138        debug_assert!(b.blocks[current].terminator.is_none(), "emitting into a sealed block");
139        b.blocks[current].instrs.push(Instruction { kind, offset });
140    }
141
142    fn emit_const(&mut self, value: Value) -> Reg {
143        let dst = self.fresh_reg();
144        self.emit(InstrKind::Const { dst, value });
145        dst
146    }
147
148    /// Seals the current block with `terminator`. If the block is already
149    /// sealed (statements after a `return`), the terminator is dropped —
150    /// that code is unreachable and lowering into a fresh floating block
151    /// keeps it out of the CFG entirely.
152    fn seal(&mut self, terminator: Terminator) {
153        let offset = self.current_offset;
154        let b = self.builder();
155        let current = b.current;
156        if b.blocks[current].terminator.is_none() {
157            b.blocks[current].terminator = Some((terminator, offset));
158        }
159    }
160
161    fn is_sealed(&mut self) -> bool {
162        let b = self.builder();
163        b.blocks[b.current].terminator.is_some()
164    }

168    /// Lowers a function/script body with the Phase 2 §4 implicit-return
169    /// rule: a trailing expression statement's register is returned;
170    /// anything else returns a `Unit` const.
171    fn lower_body(&mut self, statements: &[ASTStatement]) {
172        let Some((last, rest)) = statements.split_last() else {
173            let unit = self.emit_const(Value::Unit);
174            self.seal(Terminator::Return(unit));
175            return;
176        };
177
178        for stmt in rest {
179            self.lower_statement(stmt);
180        }
181
182        // A trailing statement after a `return` is unreachable: divert it
183        // into a detached block, same as lower_statement does.
184        if self.is_sealed() {
185            let detached = self.new_block();
186            self.switch_to(detached);
187        }
188
189        if let ASTStatementKind::Expression(expr) = &last.kind {
190            let result = self.lower_expression(expr);
191            self.seal(Terminator::Return(result));
192        } else {
193            self.lower_statement(last);
194            if !self.is_sealed() {
195                let unit = self.emit_const(Value::Unit);
196                self.seal(Terminator::Return(unit));
197            }
198        }
199    }
200
201    fn lower_statements(&mut self, statements: &[ASTStatement]) {
202        for stmt in statements {
203            self.lower_statement(stmt);
204        }
205    }
206
207    fn lower_statement(&mut self, stmt: &ASTStatement) {
208        // Unreachable code after a `return` in the same body: lower it into
209        // a detached block (no predecessors) rather than corrupting the
210        // sealed one; unreachable-block elimination will drop it.
211        if self.is_sealed() {
212            let detached = self.new_block();
213            self.switch_to(detached);
214        }
215        match &stmt.kind {
216            ASTStatementKind::Expression(expr) => {
217                self.lower_expression(expr); // result register goes unused
218            }
219            ASTStatementKind::VariableDeclaration(decl) => {
220                let value = self.lower_expression(&decl.initializer);
221                if let Some(token) = &decl.name_span {
222                    self.current_offset = token.span.start as u32;
223                }
224                let binding = decl.binding.get().expect("Resolver guarantees every declaration is resolved");
225                self.emit_store(binding, value);
226            }
227            ASTStatementKind::Assignment(assign) => {
228                let mut value = self.lower_expression(&assign.value);
229                if let Some(token) = &assign.name_span {
230                    self.current_offset = token.span.start as u32;
231                }
232                if assign.needs_float_widen.get() {
233                    let widened = self.fresh_reg();
234                    self.emit(InstrKind::ToFloat { dst: widened, src: value });
235                    value = widened;
236                }
237                let binding = assign.binding.get().expect("Resolver guarantees every assignment target is resolved");
238                self.emit_store(binding, value);
239            }
240            ASTStatementKind::Block(statements) => self.lower_statements(statements),
241            ASTStatementKind::IfStatement(if_stmt) => self.lower_if(if_stmt),
242            ASTStatementKind::FunctionDeclaration(func) => self.lower_function(func),
243            ASTStatementKind::ReturnStatement(ret) => {
244                let value = self.lower_expression(&ret.value);
245                self.seal(Terminator::Return(value));
246            }
247        }
248    }
249
250    fn lower_if(&mut self, if_stmt: &ASTIfStatement) {
251        let cond = self.lower_expression(&if_stmt.condition);
252        let then_block = self.new_block();
253        let else_block = self.new_block();
254        self.seal(Terminator::Branch { cond, then_block, else_block });
255
256        self.switch_to(then_block);
257        self.lower_statements(&if_stmt.then_branch);
258        let then_end_sealed = self.is_sealed();
259        let then_end = BlockId(self.builder().current as u32);
260
261        self.switch_to(else_block);
262        if let Some(else_branch) = &if_stmt.else_branch {
263            self.lower_statements(else_branch);
264        }
265        let else_end_sealed = self.is_sealed();
266        let else_end = BlockId(self.builder().current as u32);
267
268        let join = self.new_block();
269        if !then_end_sealed {
270            self.switch_to(then_end);
271            self.seal(Terminator::Jump(join));
272        }
273        if !else_end_sealed {
274            self.switch_to(else_end);
275            self.seal(Terminator::Jump(join));
276        }
277        self.switch_to(join);
278    }
279
280    fn lower_function(&mut self, func: &ASTFunctionDeclaration) {
281        let id = func.function_id.get().expect("Resolver assigns every function an id");
282        let frame_size = func.frame_size.get().expect("Resolver computes every function's frame size");
283        // Parameter slots are 0..arity in order (asserted in the bytecode
284        // compiler); IR functions receive arguments directly in those slots,
285        // same contract as the VM.
286        let saved_offset = self.current_offset;
287        self.push_function(func.name.clone(), frame_size);
288        self.lower_body(&func.body);
289        let function = self.pop_function();
290        self.functions[id.0 as usize] = Some(function);
291        self.current_offset = saved_offset;
292    }
293
294    fn emit_store(&mut self, binding: ResolvedBinding, src: Reg) {
295        match binding {
296            ResolvedBinding::Local(slot) => self.emit(InstrKind::StoreLocal { slot, src }),
297            ResolvedBinding::Global(slot) => self.emit(InstrKind::StoreGlobal { slot, src }),
298        }
299    }
300
301    fn emit_load(&mut self, binding: ResolvedBinding) -> Reg {
302        let dst = self.fresh_reg();
303        match binding {
304            ResolvedBinding::Local(slot) => self.emit(InstrKind::LoadLocal { dst, slot }),
305            ResolvedBinding::Global(slot) => self.emit(InstrKind::LoadGlobal { dst, slot }),
306        }
307        dst
308    }

312    fn lower_expression(&mut self, expr: &ASTExpression) -> Reg {
313        match &expr.kind {
314            ASTExpressionKind::Number(n) => self.emit_const(n.value.clone()),
315            ASTExpressionKind::Paranthesized(p) => self.lower_expression(&p.expression),
316            ASTExpressionKind::Identifier(ident) => {
317                if let Some(token) = &ident.token {
318                    self.current_offset = token.span.start as u32;
319                }
320                let binding = ident.binding.get().expect("Resolver guarantees every identifier is resolved");
321                self.emit_load(binding)
322            }
323            ASTExpressionKind::Binary(bin) => self.lower_binary(bin),
324            ASTExpressionKind::Unary(u) => {
325                self.current_offset = u.operator.token.span.start as u32;
326                match u.operator.kind {
327                    ASTUnaryOperatorKind::Increment | ASTUnaryOperatorKind::Decrement => {
328                        let is_inc = matches!(u.operator.kind, ASTUnaryOperatorKind::Increment);
329                        self.lower_inc_dec(&u.operand, is_inc, false)
330                    }
331                    // Runtime passthrough — nothing to emit.
332                    ASTUnaryOperatorKind::Plus => self.lower_expression(&u.operand),
333                    ASTUnaryOperatorKind::Minus => {
334                        let src = self.lower_expression(&u.operand);
335                        let can_error = !is_static_numeric(&u.operand);
336                        let dst = self.fresh_reg();
337                        self.current_offset = u.operator.token.span.start as u32;
338                        self.emit(InstrKind::Negate { dst, src, can_error });
339                        dst
340                    }
341                    ASTUnaryOperatorKind::LogicalNot => {
342                        let src = self.lower_expression(&u.operand);
343                        let dst = self.fresh_reg();
344                        self.current_offset = u.operator.token.span.start as u32;
345                        self.emit(InstrKind::Not { dst, src });
346                        dst
347                    }
348                }
349            }
350            ASTExpressionKind::PostfixUnary(p) => {
351                self.current_offset = p.operator.token.span.start as u32;
352                let is_inc = matches!(p.operator.kind, ASTUnaryOperatorKind::Increment);
353                self.lower_inc_dec(&p.operand, is_inc, true)
354            }
355            ASTExpressionKind::FunctionCall(call) => self.lower_call(call),
356        }
357    }
358
359    fn lower_binary(&mut self, bin: &ASTBinaryExpression) -> Reg {
360        let op_offset = bin.operator.token.span.start as u32;
361        match bin.operator.kind {
362            // Short-circuit: the right operand only evaluates in its arm.
363            // The result register is written in both arms (the IR's one
364            // deliberate deviation from single-assignment) and holds a real
365            // Boolean, matching both existing backends' coercion.
366            ASTBinaryOperatorKind::LogicalAnd | ASTBinaryOperatorKind::LogicalOr => {
367                let is_and = matches!(bin.operator.kind, ASTBinaryOperatorKind::LogicalAnd);
368                let lhs = self.lower_expression(&bin.left);
369                self.current_offset = op_offset;
370                let result = self.fresh_reg();
371                let rhs_block = self.new_block();
372                let short_block = self.new_block();
373                let (then_block, else_block) = if is_and {
374                    (rhs_block, short_block) // truthy lhs -> evaluate rhs
375                } else {
376                    (short_block, rhs_block) // truthy lhs -> short-circuit true
377                };
378                self.seal(Terminator::Branch { cond: lhs, then_block, else_block });
379
380                self.switch_to(rhs_block);
381                let rhs = self.lower_expression(&bin.right);
382                self.current_offset = op_offset;
383                self.emit(InstrKind::ToBool { dst: result, src: rhs });
384                let rhs_end = BlockId(self.builder().current as u32);
385
386                self.switch_to(short_block);
387                self.emit(InstrKind::Const { dst: result, value: Value::Boolean(!is_and) });
388
389                let join = self.new_block();
390                self.seal(Terminator::Jump(join));
391                self.switch_to(rhs_end);
392                self.seal(Terminator::Jump(join));
393                self.switch_to(join);
394                result
395            }
396            op => {
397                let lhs = self.lower_expression(&bin.left);
398                let rhs = self.lower_expression(&bin.right);
399                let can_error = binary_can_error(op, &bin.left, &bin.right);
400                let dst = self.fresh_reg();
401                self.current_offset = op_offset;
402                self.emit(InstrKind::Binary { dst, op, lhs, rhs, can_error });
403                dst
404            }
405        }
406    }

408    /// Same desugaring as the bytecode compiler: load, `+`/`-` 1, store;
409    /// prefix yields the new value, postfix the original.
410    fn lower_inc_dec(&mut self, operand: &ASTExpression, is_inc: bool, is_postfix: bool) -> Reg {
411        let ASTExpressionKind::Identifier(ident) = &operand.kind else {
412            unreachable!("Resolver guarantees increment/decrement operand is an identifier");
413        };
414        let binding = ident.binding.get().expect("Resolver guarantees this identifier is resolved");
415
416        let old = self.emit_load(binding);
417        let one = self.emit_const(Value::Integer(1));
418        let new = self.fresh_reg();
419        let op = if is_inc { ASTBinaryOperatorKind::Plus } else { ASTBinaryOperatorKind::Minus };
420        let can_error = !is_static_numeric(operand);
421        self.emit(InstrKind::Binary { dst: new, op, lhs: old, rhs: one, can_error });
422        self.emit_store(binding, new);
423        if is_postfix {
424            old
425        } else {
426            new
427        }
428    }

430    fn lower_call(&mut self, call: &ASTFunctionCallExpression) -> Reg {
431        if let Some(token) = &call.token {
432            self.current_offset = token.span.start as u32;
433        }
434        let call_offset = self.current_offset;
435        let args: Vec<Reg> = call.arguments.iter().map(|arg| self.lower_expression(arg)).collect();
436        let dst = self.fresh_reg();
437        self.current_offset = call_offset;
438        match call.resolved_call.get().expect("Resolver guarantees every call is resolved") {
439            crate::ast::resolver::ResolvedCall::User(id) => {
440                self.emit(InstrKind::Call { dst, function: id.0, args });
441            }
442            crate::ast::resolver::ResolvedCall::Builtin(builtin) => {
443                self.emit(InstrKind::CallBuiltin { dst, builtin, args });
444            }
445        }
446        dst
447    }

450/// `true` iff the expression's Resolver-inferred type is statically numeric
451/// — the operand-side half of the shared can-this-error predicate. An
452/// unresolved or `Any` type means runtime coercion can fail.
453fn is_static_numeric(expr: &ASTExpression) -> bool {
454    matches!(expr.resolved_type.get(), Some(DataType::Integer) | Some(DataType::Float))
455}
456
457fn is_static_concrete(expr: &ASTExpression) -> bool {
458    !matches!(expr.resolved_type.get(), None | Some(DataType::Any))
459}

461/// The shared purity fact (design doc §5.2): a binary op can raise a runtime
462/// error iff it's inherently partial (`/`/`%` by zero, `**` overflow) or
463/// either operand is dynamically typed. Statically-typed operands of any
464/// other op already passed the Resolver's legality check and cannot fail.
465fn binary_can_error(op: ASTBinaryOperatorKind, left: &ASTExpression, right: &ASTExpression) -> bool {
466    use ASTBinaryOperatorKind::*;
467    matches!(op, Divide | Modulo | Exponentiation)
468        || !is_static_concrete(left)
469        || !is_static_concrete(right)
