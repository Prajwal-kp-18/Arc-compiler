1//! # Disassembler
2//!
3//! A pretty-printer for [`Chunk`]s — the tool for eyeballing generated
4//! bytecode while hand-tracing programs, and for debugging the VM once
5//! Phase 3 exists. Wired up behind `arc --dump-bytecode file.arc`.
6
7use crate::ast::resolver::BuiltinFn;
8use crate::bytecode::chunk::Chunk;
9use crate::bytecode::compiler::CompiledProgram;
10use crate::bytecode::opcode::OpCode;
11
12/// Disassembles every chunk in a compiled program: the script, then each
13/// function in `FunctionId` order.
14pub fn disassemble_program(program: &CompiledProgram) -> String {
15    let mut out = disassemble_chunk(&program.script, "script");
16    for (id, function) in program.functions.iter().enumerate() {
17        out.push('\n');
18        out.push_str(&disassemble_chunk(&function.chunk, &format!("fn#{} {}", id, function.name)));
19    }
20    out
21}
22
23pub fn disassemble_chunk(chunk: &Chunk, name: &str) -> String {
24    let mut out = format!("== {} ==\n", name);
25    let mut offset = 0;
26    while offset < chunk.code.len() {
27        offset = disassemble_instruction(chunk, offset, &mut out);
28    }
29    out
30}
31
32/// Disassembles one instruction at `offset`, appending its line to `out`,
33/// and returns the offset of the next instruction.
34fn disassemble_instruction(chunk: &Chunk, offset: usize, out: &mut String) -> usize {
35    let src_offset = chunk.offsets[offset];
36    let op = OpCode::from_byte(chunk.code[offset]).expect("invalid opcode byte in chunk");
37    let prefix = format!("{:04}  src:{:<5}", offset, src_offset);
38
39    match op {
40        OpCode::Constant => {
41            let idx = chunk.read_u16(offset + 1);
42            out.push_str(&format!("{} {:<16} {:4} '{:?}'\n", prefix, op.mnemonic(), idx, chunk.constants[idx as usize]));
43            offset + 3
44        }
45        OpCode::GetLocal | OpCode::SetLocal | OpCode::GetGlobal | OpCode::SetGlobal => {
46            let slot = chunk.read_u16(offset + 1);
47            out.push_str(&format!("{} {:<16} {:4}\n", prefix, op.mnemonic(), slot));
48            offset + 3
49        }
50        OpCode::Jump | OpCode::JumpIfFalse => {
51            let jump = chunk.read_u16(offset + 1);
52            let target = offset + 3 + jump as usize;
53            out.push_str(&format!("{} {:<16} {:4} -> {:04}\n", prefix, op.mnemonic(), jump, target));
54            offset + 3
55        }
56        OpCode::Loop => {
57            let distance = chunk.read_u16(offset + 1);
58            let target = offset + 3 - distance as usize;
59            out.push_str(&format!("{} {:<16} {:4} -> {:04}\n", prefix, op.mnemonic(), distance, target));
60            offset + 3
61        }
62        OpCode::Call => {
63            let function_id = chunk.read_u16(offset + 1);
64            let argc = chunk.code[offset + 3];
65            out.push_str(&format!("{} {:<16} fn#{} argc={}\n", prefix, op.mnemonic(), function_id, argc));
66            offset + 4
67        }
68        OpCode::CallBuiltin => {
69            let builtin_id = chunk.code[offset + 1];
70            let argc = chunk.code[offset + 2];
71            let name = BuiltinFn::from_u8(builtin_id).map(BuiltinFn::name).unwrap_or("?");
72            out.push_str(&format!("{} {:<16} {:<8} argc={}\n", prefix, op.mnemonic(), name, argc));
73            offset + 3
74        }
75        // No operand.
76        OpCode::True
77        | OpCode::False
78        | OpCode::Pop
79        | OpCode::ToFloat
80        | OpCode::Add
81        | OpCode::Subtract
82        | OpCode::Multiply
83        | OpCode::Divide
84        | OpCode::Modulo
85        | OpCode::Power
86        | OpCode::BitAnd
87        | OpCode::BitOr
88        | OpCode::BitXor
89        | OpCode::ShiftLeft
90        | OpCode::ShiftRight
91        | OpCode::Equal
92        | OpCode::NotEqual
93        | OpCode::Less
94        | OpCode::Greater
95        | OpCode::LessEqual
96        | OpCode::GreaterEqual
97        | OpCode::Negate
98        | OpCode::Not
99        | OpCode::Return => {
100            out.push_str(&format!("{} {}\n", prefix, op.mnemonic()));
101            offset + 1
102        }
103    }
104}
