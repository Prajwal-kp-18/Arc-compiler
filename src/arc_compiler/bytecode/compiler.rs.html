1//! # Bytecode Compiler
2//!
3//! Lowers a Resolver-annotated AST into a [`CompiledProgram`] of [`Chunk`]s.
4//! No execution happens here — see the Phase 2 design doc
5//! (`local-notes/phase2-bytecode-design.md`) for the full rationale.
6//!
7//! Like the evaluator, this is infallible: it trusts the Resolver's
8//! guarantees completely and reads `SlotIndex`/`FunctionId`/`ResolvedBinding`/
9//! `ResolvedCall` straight off the AST's `Cell` fields instead of
10//! re-deriving scope.
11
12use crate::ast::resolver::{BuiltinFn, ResolvedBinding, ResolvedCall};
13use crate::ast::types::Value;
14use crate::ast::{
15    Ast, ASTBinaryExpression, ASTBinaryOperatorKind, ASTExpression, ASTExpressionKind,
16    ASTFunctionCallExpression, ASTFunctionDeclaration, ASTIfStatement, ASTPostfixUnaryExpression,
17    ASTStatement, ASTStatementKind, ASTUnaryExpression, ASTUnaryOperatorKind,
18};
19use crate::bytecode::chunk::Chunk;
20use crate::bytecode::opcode::OpCode;
21
22/// The output of a compile pass: the top-level "script" chunk, plus one
23/// compiled function per `FunctionId`, indexed by `FunctionId.0`.
24pub struct CompiledProgram {
25    pub script: Chunk,
26    pub functions: Vec<CompiledFunction>,
27    /// Local slots the *script* frame needs. Zero for directly-compiled
28    /// programs (script variables are globals); non-zero when the IR
29    /// pipeline spills the script's virtual registers to frame slots.
30    pub script_frame_size: u32,
31}
32
33/// A function's chunk plus the metadata the VM needs to call it. No arity
34/// field: the `Call` opcode carries its own argument count and the Resolver
35/// already rejected arity mismatches statically.
36pub struct CompiledFunction {
37    pub name: String,
38    pub frame_size: u32,
39    pub chunk: Chunk,
40}
41
42impl CompiledFunction {
43    fn placeholder() -> Self {
44        Self { name: String::new(), frame_size: 0, chunk: Chunk::new() }
45    }
46}
47
48pub struct BytecodeCompiler {
49    functions: Vec<CompiledFunction>,
50    /// In-progress chunks; the top is the one currently being written to.
51    /// Depth > 1 only while compiling a function nested inside another.
52    chunk_stack: Vec<Chunk>,
53    /// Source byte-offset of the most recently seen token, reused for AST
54    /// nodes that don't carry their own (e.g. number literals) so
55    /// disassembly still points somewhere close to the right place.
56    current_offset: u32,
57}
58
59impl BytecodeCompiler {
60    pub fn new(function_count: u32) -> Self {
61        Self {
62            functions: (0..function_count).map(|_| CompiledFunction::placeholder()).collect(),
63            chunk_stack: Vec::new(),
64            current_offset: 0,
65        }
66    }
67
68    pub fn compile(mut self, ast: &Ast) -> CompiledProgram {
69        self.chunk_stack.push(Chunk::new());
70        self.compile_body(&ast.statements);
71        let script = self.chunk_stack.pop().expect("script chunk was just pushed");
72        CompiledProgram { script, functions: self.functions, script_frame_size: 0 }
73    }
74
75    fn current(&mut self) -> &mut Chunk {
76        self.chunk_stack.last_mut().expect("always compiling inside some chunk")
77    }
78
79    fn emit(&mut self, op: OpCode, offset: u32) {
80        self.current().write_op(op, offset);
81    }
82
83    fn emit_constant(&mut self, value: Value, offset: u32) {
84        let idx = self.current().add_constant(value);
85        self.current().write_op(OpCode::Constant, offset);
86        self.current().write_u16(idx, offset);
87    }
88
89    fn slot_u16(index: u32) -> u16 {
90        u16::try_from(index).expect("slot index exceeded u16::MAX")
91    }
92
93    fn emit_get(&mut self, binding: ResolvedBinding, offset: u32) {
94        let (op, slot) = match binding {
95            ResolvedBinding::Local(s) => (OpCode::GetLocal, s.0),
96            ResolvedBinding::Global(s) => (OpCode::GetGlobal, s.0),
97        };
98        self.current().write_op(op, offset);
99        self.current().write_u16(Self::slot_u16(slot), offset);
100    }
101
102    fn emit_set(&mut self, binding: ResolvedBinding, offset: u32) {
103        let (op, slot) = match binding {
104            ResolvedBinding::Local(s) => (OpCode::SetLocal, s.0),
105            ResolvedBinding::Global(s) => (OpCode::SetGlobal, s.0),
106        };
107        self.current().write_op(op, offset);
108        self.current().write_u16(Self::slot_u16(slot), offset);
109    }
110
111    /// Pushed where there's genuinely no value to give: an implicit return
112    /// with no trailing expression statement (design doc §4). `Unit` (not
113    /// the Phase 2 `Integer(0)` placeholder) so misusing the "result" is a
114    /// runtime error in the VM, matching the tree-walker's behavior.
115    fn emit_placeholder(&mut self, offset: u32) {
116        self.emit_constant(Value::Unit, offset);
117    }
118
119    // -- statements -----------------------------------------------------------
120
121    /// Compiles a function/script body: every statement is stack-neutral
122    /// except (optionally) the last. If the last statement is an expression
123    /// statement, its value becomes the implicit return (no trailing `POP`);
124    /// otherwise a placeholder is pushed. Either way, always ends in
125    /// `OP_RETURN` — every chunk needs one, explicit `return` or not (see
126    /// design doc §4).
127    fn compile_body(&mut self, statements: &[ASTStatement]) {
128        let Some((last, rest)) = statements.split_last() else {
129            let offset = self.current_offset;
130            self.emit_placeholder(offset);
131            self.emit(OpCode::Return, offset);
132            return;
133        };
134
135        for stmt in rest {
136            self.compile_statement(stmt);
137        }
138
139        if let ASTStatementKind::Expression(expr) = &last.kind {
140            self.compile_expression(expr);
141        } else {
142            self.compile_statement(last);
143            self.emit_placeholder(self.current_offset);
144        }
145        self.emit(OpCode::Return, self.current_offset);
146    }
147
148    fn compile_statements(&mut self, statements: &[ASTStatement]) {
149        for stmt in statements {
150            self.compile_statement(stmt);
151        }
152    }
153
154    fn compile_statement(&mut self, stmt: &ASTStatement) {
155        match &stmt.kind {
156            ASTStatementKind::Expression(expr) => {
157                self.compile_expression(expr);
158                self.emit(OpCode::Pop, self.current_offset);
159            }
160            ASTStatementKind::VariableDeclaration(decl) => {
161                self.compile_expression(&decl.initializer);
162                if let Some(token) = &decl.name_span {
163                    self.current_offset = token.span.start as u32;
164                }
165                let binding = decl.binding.get().expect("Resolver guarantees every declaration is resolved");
166                self.emit_set(binding, self.current_offset);
167                self.emit(OpCode::Pop, self.current_offset);
168            }
169            ASTStatementKind::Assignment(assign) => {
170                self.compile_expression(&assign.value);
171                if let Some(token) = &assign.name_span {
172                    self.current_offset = token.span.start as u32;
173                }
174                if assign.needs_float_widen.get() {
175                    self.emit(OpCode::ToFloat, self.current_offset);
176                }
177                let binding = assign.binding.get().expect("Resolver guarantees every assignment target is resolved");
178                self.emit_set(binding, self.current_offset);
179                self.emit(OpCode::Pop, self.current_offset);
180            }
181            ASTStatementKind::Block(statements) => self.compile_statements(statements),
182            ASTStatementKind::IfStatement(if_stmt) => self.compile_if(if_stmt),
183            ASTStatementKind::FunctionDeclaration(func) => self.compile_function(func),
184            ASTStatementKind::ReturnStatement(ret) => {
185                self.compile_expression(&ret.value);
186                self.emit(OpCode::Return, self.current_offset);
187            }
188        }
189    }
190
191    fn compile_if(&mut self, if_stmt: &ASTIfStatement) {
192        self.compile_expression(&if_stmt.condition);
193        let offset = self.current_offset;
194
195        let then_jump = self.current().emit_jump(OpCode::JumpIfFalse, offset);
196        self.emit(OpCode::Pop, offset);
197        self.compile_statements(&if_stmt.then_branch);
198        let else_jump = self.current().emit_jump(OpCode::Jump, offset);
199
200        self.current().patch_jump(then_jump);
201        self.emit(OpCode::Pop, offset);
202        if let Some(else_branch) = &if_stmt.else_branch {
203            self.compile_statements(else_branch);
204        }
205
206        self.current().patch_jump(else_jump);
207    }
208
209    fn compile_function(&mut self, func: &ASTFunctionDeclaration) {
210        let id = func.function_id.get().expect("Resolver assigns every function an id");
211
212        // The VM pops arguments straight into frame slots 0..arity — valid
213        // because the Resolver declares parameters first in a fresh frame,
214        // so their slots are always 0, 1, 2, ... in order.
215        for (i, param) in func.parameters.iter().enumerate() {
216            let slot = param.slot.get().expect("Resolver assigns every parameter a slot");
217            debug_assert_eq!(slot.0 as usize, i, "parameter slots must be sequential from 0");
218        }
219
220        self.chunk_stack.push(Chunk::new());
221        self.compile_body(&func.body);
222        let chunk = self.chunk_stack.pop().expect("function chunk was just pushed");
223        self.functions[id.0 as usize] = CompiledFunction {
224            name: func.name.clone(),
225            frame_size: func.frame_size.get().expect("Resolver computes every function's frame size"),
226            chunk,
227        };
228    }
229
230    // -- expressions ------------------------------------------------------------
231
232    fn compile_expression(&mut self, expr: &ASTExpression) {
233        match &expr.kind {
234            ASTExpressionKind::Number(n) => {
235                let offset = self.current_offset;
236                match n.value {
237                    Value::Boolean(true) => self.emit(OpCode::True, offset),
238                    Value::Boolean(false) => self.emit(OpCode::False, offset),
239                    ref v => self.emit_constant(v.clone(), offset),
240                }
241            }
242            ASTExpressionKind::Binary(bin) => self.compile_binary(bin),
243            ASTExpressionKind::Paranthesized(p) => self.compile_expression(&p.expression),
244            ASTExpressionKind::Unary(u) => self.compile_unary(u),
245            ASTExpressionKind::PostfixUnary(p) => self.compile_postfix(p),
246            ASTExpressionKind::Identifier(ident) => {
247                if let Some(token) = &ident.token {
248                    self.current_offset = token.span.start as u32;
249                }
250                let binding = ident.binding.get().expect("Resolver guarantees every identifier is resolved");
251                self.emit_get(binding, self.current_offset);
252            }
253            ASTExpressionKind::FunctionCall(call) => self.compile_call(call),
254        }
255    }
256
257    fn binary_opcode(kind: &ASTBinaryOperatorKind) -> OpCode {
258        use ASTBinaryOperatorKind::*;
259        match kind {
260            Plus => OpCode::Add,
261            Minus => OpCode::Subtract,
262            Multiply => OpCode::Multiply,
263            Divide => OpCode::Divide,
264            Modulo => OpCode::Modulo,
265            Exponentiation => OpCode::Power,
266            BitwiseAnd => OpCode::BitAnd,
267            BitwiseOr => OpCode::BitOr,
268            BitwiseXor => OpCode::BitXor,
269            LeftShift => OpCode::ShiftLeft,
270            RightShift => OpCode::ShiftRight,
271            Equal => OpCode::Equal,
272            NotEqual => OpCode::NotEqual,
273            Less => OpCode::Less,
274            Greater => OpCode::Greater,
275            LessEqual => OpCode::LessEqual,
276            GreaterEqual => OpCode::GreaterEqual,
277            LogicalAnd | LogicalOr => unreachable!("handled directly in compile_binary"),
278        }
279    }
280
281    /// `&&`/`||` compile to jumps + double-`NOT` rather than dedicated
282    /// opcodes: Arc's logical operators always coerce their result to a real
283    /// `Boolean` (unlike Lox-style "return whichever operand"), and `NOT NOT`
284    /// performs exactly that coercion on whichever value survives on the
285    /// stack. See design doc §3.
286    fn compile_binary(&mut self, bin: &ASTBinaryExpression) {
287        self.current_offset = bin.operator.token.span.start as u32;
288        let offset = self.current_offset;
289
290        match bin.operator.kind {
291            ASTBinaryOperatorKind::LogicalAnd => {
292                self.compile_expression(&bin.left);
293                let end_jump = self.current().emit_jump(OpCode::JumpIfFalse, offset);
294                self.emit(OpCode::Pop, offset);
295                self.compile_expression(&bin.right);
296                self.current().patch_jump(end_jump);
297                self.emit(OpCode::Not, offset);
298                self.emit(OpCode::Not, offset);
299            }
300            ASTBinaryOperatorKind::LogicalOr => {
301                self.compile_expression(&bin.left);
302                let else_jump = self.current().emit_jump(OpCode::JumpIfFalse, offset);
303                let end_jump = self.current().emit_jump(OpCode::Jump, offset);
304                self.current().patch_jump(else_jump);
305                self.emit(OpCode::Pop, offset);
306                self.compile_expression(&bin.right);
307                self.current().patch_jump(end_jump);
308                self.emit(OpCode::Not, offset);
309                self.emit(OpCode::Not, offset);
310            }
311            ref kind => {
312                self.compile_expression(&bin.left);
313                self.compile_expression(&bin.right);
314                self.emit(Self::binary_opcode(kind), offset);
315            }
316        }
317    }
318
319    fn compile_unary(&mut self, u: &ASTUnaryExpression) {
320        self.current_offset = u.operator.token.span.start as u32;
321        let offset = self.current_offset;
322        match u.operator.kind {
323            ASTUnaryOperatorKind::Increment | ASTUnaryOperatorKind::Decrement => {
324                self.compile_inc_dec(&u.operand, matches!(u.operator.kind, ASTUnaryOperatorKind::Increment), false, offset);
325            }
326            // Runtime passthrough: valid for every type, never errors, so
327            // there's nothing to emit at all.
328            ASTUnaryOperatorKind::Plus => self.compile_expression(&u.operand),
329            ASTUnaryOperatorKind::Minus => {
330                self.compile_expression(&u.operand);
331                self.emit(OpCode::Negate, offset);
332            }
333            ASTUnaryOperatorKind::LogicalNot => {
334                self.compile_expression(&u.operand);
335                self.emit(OpCode::Not, offset);
336            }
337        }
338    }
339
340    fn compile_postfix(&mut self, p: &ASTPostfixUnaryExpression) {
341        self.current_offset = p.operator.token.span.start as u32;
342        let offset = self.current_offset;
343        let is_increment = matches!(p.operator.kind, ASTUnaryOperatorKind::Increment);
344        self.compile_inc_dec(&p.operand, is_increment, true, offset);
345    }
346
347    /// Desugars `++x`/`--x`/`x++`/`x--` into `GET`/`CONSTANT 1`/`ADD`-or-
348    /// `SUBTRACT`/`SET` (plus a `POP` for postfix, to discard the new value
349    /// and keep the old one), reusing the generic runtime arithmetic instead
350    /// of dedicated opcodes.
351    ///
352    /// ponytail: this means a String/Boolean value flowing through an
353    /// untyped (`Any`) variable that gets incremented will error as "Cannot
354    /// add" rather than "Cannot increment" once a VM exists. Every *typed*
355    /// case is already rejected at resolve time, so only that narrow
356    /// dynamic-fallback sliver is affected — an accepted trade-off, not an
357    /// oversight (design doc §3).
358    fn compile_inc_dec(&mut self, operand: &ASTExpression, is_increment: bool, is_postfix: bool, offset: u32) {
359        let ASTExpressionKind::Identifier(ident) = &operand.kind else {
360            unreachable!("Resolver guarantees increment/decrement operand is an identifier");
361        };
362        let binding = ident.binding.get().expect("Resolver guarantees this identifier is resolved");
363
364        if is_postfix {
365            self.emit_get(binding, offset); // old value: kept as the final result
366        }
367        self.emit_get(binding, offset);
368        self.emit_constant(Value::Integer(1), offset);
369        self.emit(if is_increment { OpCode::Add } else { OpCode::Subtract }, offset);
370        self.emit_set(binding, offset); // leaves the new value on the stack
371        if is_postfix {
372            self.emit(OpCode::Pop, offset); // discard new value; old value remains
373        }
374    }
375
376    fn compile_call(&mut self, call: &ASTFunctionCallExpression) {
377        if let Some(token) = &call.token {
378            self.current_offset = token.span.start as u32;
379        }
380        let offset = self.current_offset;
381
382        for arg in &call.arguments {
383            self.compile_expression(arg);
384        }
385        let argc = u8::try_from(call.arguments.len()).expect("more than 255 arguments in a single call");
386
387        match call.resolved_call.get().expect("Resolver guarantees every call is resolved") {
388            ResolvedCall::User(id) => {
389                self.emit(OpCode::Call, offset);
390                self.current().write_u16(Self::slot_u16(id.0), offset);
391                self.current().write_u8(argc, offset);
392            }
393            ResolvedCall::Builtin(builtin) => {
394                let op = match builtin {
395                    BuiltinFn::Print => OpCode::Print,
396                    BuiltinFn::Max => OpCode::Max,
397                    BuiltinFn::Min => OpCode::Min,
398                };
399                self.emit(op, offset);
400                self.current().write_u8(argc, offset);
401            }
402        }
403    }

414    /// Lexes, parses, resolves, and compiles a program, panicking on any
415    /// parse/resolve error (these tests are about the bytecode compiler,
416    /// not upstream diagnostics).
417    fn compile(source: &str) -> CompiledProgram {
418        let mut lexer = Lexer::new(source);
419        let mut tokens = Vec::new();
420        while let Some(token) = lexer.next_token() {
421            tokens.push(token);
422        }
423        let mut parser = Parser::new(tokens);
424        let mut ast = Ast::new();
425        while let Some(stmt) = parser.next_statement() {
426            ast.add_statement(stmt);
427        }
428        assert!(
429            parser.diagnostics.is_empty(),
430            "parse errors: {:?}",
431            parser.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>()
432        );
433
434        let mut resolver = Resolver::new();
435        resolver.resolve(&ast);
436        assert!(
437            !resolver.has_errors(),
438            "resolve errors: {:?}",
439            resolver.diagnostics.iter().map(|d| &d.message).collect::<Vec<_>>()
440        );
441
442        BytecodeCompiler::new(resolver.function_count()).compile(&ast)
443    }

445    fn op_lines(text: &str) -> Vec<&str> {
446        text.lines().filter(|l| l.contains("OP_")).collect()
447    }

449    #[test]
450    fn test_arithmetic_compiles_and_disassembles() {
451        let program = compile("1 + 2 * 3;");
452        let text = disassemble_chunk(&program.script, "script");
453        assert!(text.contains("OP_CONSTANT"));
454        assert!(text.contains("OP_MULTIPLY"));
455        assert!(text.contains("OP_ADD"));
456        assert!(text.contains("OP_RETURN"));
457        // Last statement is an expression statement -> falls through to
458        // RETURN with no trailing POP (design doc §4).
459        assert!(!text.contains("OP_POP"));
460    }

525    #[test]
526    fn test_print_of_print_stack_effect() {
527        // The narrowest case Arc's grammar allows for a call's result being
528        // consumed by another call: print(print("x")). The outer PRINT must
529        // consume exactly the inner call's placeholder result — no extra
530        // push should appear between the two PRINT instructions (design doc §5).
531        let program = compile(r#"print(print("x"));"#);
532        let text = disassemble_chunk(&program.script, "script");
533        let lines = op_lines(&text);
534        assert_eq!(lines.len(), 4, "{:#?}", lines);
535        assert!(lines[0].contains("OP_CONSTANT"));
536        assert!(lines[1].contains("OP_PRINT"));
537        assert!(lines[2].contains("OP_PRINT"));
538        assert!(lines[3].contains("OP_RETURN"));
